; After the Dam: keeps the Securitron escort's order (2026-11-19), every 5 seconds after the ending
; and once right after each load (Tick.txt). Told to wait or go home, it gets its package back: after
; a load (in case the save didn't keep it), and whenever it's near you doing something else (its own
; follow package took over again).
int iState
ref rEsc
ref rPack
ref rFn

begin Function {}
	let iState := player.AuxVarGetFlt "_adEsc"
	let rEsc := BuildRef 0 1397979	; VHDSecuritronEscort01REF
	if eval iState < 2 || (IsFormValid rEsc) == 0
		player.AuxVarSetFlt "*_adUp" 1
		return
	endif
	if eval (rEsc.GetDead) || (rEsc.GetDisabled)
		player.AuxVarSetFlt "*_adUp" 1
		return
	endif
	let rFn := CompileScript "AfterDam/EscortSet.txt"
	if eval (player.AuxVarGetFlt "*_adUp") == 0
		player.AuxVarSetFlt "*_adUp" 1
		call rFn iState 0
		return
	endif
	if eval (rEsc.HasLoaded3D) == 0
		return
	endif
	if iState == 2
		let rPack := BuildRef 0 1216788	; DefaultSandboxCurrentLocation256NoSleepNoWander
	else
		let rPack := BuildRef 0 1190868	; DefaultSandboxEditorLocation256NoSleep
	endif
	if eval (rEsc.GetIsCurrentPackage rPack) == 0
		call rFn iState 0
	endif
end
