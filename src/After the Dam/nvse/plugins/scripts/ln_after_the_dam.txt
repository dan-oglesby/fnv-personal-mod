; After the Dam (2026-11-17): play goes on after the ending. When the end slides finish, instead of
; fading to the credits and quitting, the game puts things back (saving, controls, the companions
; who were with you) and you wake on the road by Boulder City, at the developers' own unused
; post-game start. Nothing else about the world changes. Scripts only, no plugin; don't use it with
; another post-ending mod.
;
; Run by JIP LN NVSE's script runner (ln_: new game and every load). Needs xNVSE and JIP LN.

let int iBad := 0
let ref rChk := CompileScript "AfterDam/Always.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "After the Dam: AfterDam/Always.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
let rChk := CompileScript "AfterDam/Tick.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "After the Dam: AfterDam/Tick.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
let rChk := CompileScript "AfterDam/Resume.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "After the Dam: AfterDam/Resume.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
if iBad > 0
	print "After the Dam: " + $iBad + " script function(s) failed to compile (named above). Everything else in it is running."
	MessageEx "After the Dam: part of it is off, a script failed to compile (see the console)"
endif
if iBad >= 0	; always (2026-11-17): each part below is registered only if its own function compiled
	; twice a second: flags 4 + 16, not in menus, not while paused (the slides are gameplay)
	let ref rTick := CompileScript "AfterDam/Tick.txt"
	let ref rTrue := CompileScript "AfterDam/Always.txt"
	let ref rStart := CompileScript "AfterDam/Start.txt"
	if eval (IsFormValid rTick) && (IsFormValid rTrue) && (IsFormValid rStart)
		CallAfterSeconds 0.05 rStart 1 rTick rTrue 0.5 20	; timer slot 0.050 s (tools/timer_slots.py)
	endif
endif
