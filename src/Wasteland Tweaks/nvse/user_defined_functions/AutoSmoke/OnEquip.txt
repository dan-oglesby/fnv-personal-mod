; Auto Smoke: xNVSE OnActorEquip handler. When the player equips the cigarette case (in the
; Pip-Boy), opens the settings once the Pip-Boy is closed.
ref rActor
ref rItem
int iPlugin
ref rCase
ref rUse
ref rPlayer

begin Function {rActor, rItem}
	let iPlugin := GetModIndex "AutoSmoke.esp"
	let rCase := BuildRef iPlugin 2048	; ASCigaretteCase
	let rPlayer := BuildRef 0 20	; 0x14 = the player
	if eval (rItem != rCase) || (rActor != rPlayer)
		return
	endif
	let rUse := CompileScript "AutoSmoke/UseCase.txt"
	CallAfterSeconds 0.2 rUse	; runs in gameplay only, so after the Pip-Boy closes
end
