; Auto Smoke: the only place that touches the Hardcore sleep-deprivation value (actor value
; SleepDeprevation, spelled that way in the game's own name table (code 75); higher = more tired, stages at 200/400/600/800/1000). Kept alone so
; that if it ever fails to compile, only nicotine alertness switches off (ln_auto_smoke.txt checks
; it separately), not the whole of Auto Smoke.
; iOp 0: returns the current value. iOp 1: changes it by fAmt (never below 0), returns the change.
int iOp
float fAmt
float fCur
float fDelta

begin Function {iOp, fAmt}
	let fCur := player.GetAV SleepDeprevation
	if iOp == 0
		SetFunctionValue fCur
		return
	endif
	let fDelta := fAmt
	if fCur + fDelta < 0
		let fDelta := 0 - fCur
	endif
	if fDelta != 0
		player.ModAV SleepDeprevation fDelta
		; report the change that actually happened, so the console line after a smoke shows
		; whether it took effect
		let fDelta := player.GetAV SleepDeprevation
		let fDelta -= fCur
	endif
	SetFunctionValue fDelta
end
