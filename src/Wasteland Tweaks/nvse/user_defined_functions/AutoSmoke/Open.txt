; Auto Smoke: the settings menu. Counts are read live when it opens. The text is built in pieces
; (a script line can be at most 512 characters).
int iPlugin
int iCigs
int iRillos
int iCigars
int iPacks
int iCartons
int iGap
string_var sOn
string_var sOnBtn
string_var sWhen
string_var sPacks
string_var sPacksBtn
string_var sNic
string_var sNicBtn
string_var sBody
ref rForm
ref rCb

begin Function {}
	let iPlugin := GetModIndex "LightUpAndSmokeThoseCigarettes_edisleado.esp"
	let rForm := BuildRef iPlugin 39432	; AidCigarette
	let iCigs := player.GetItemCount rForm
	let rForm := BuildRef iPlugin 56462	; AidCigarillo
	let iRillos := player.GetItemCount rForm
	let rForm := BuildRef iPlugin 56448	; AidCigar
	let iCigars := player.GetItemCount rForm
	let rForm := BuildRef iPlugin 37960	; AidCigarettePack
	let iPacks := player.GetItemCount rForm
	let rForm := BuildRef iPlugin 37226	; AidCigaretteCarton
	let iCartons := player.GetItemCount rForm

	let sOn := "off"
	let sOnBtn := "Turn auto-smoke on"
	if eval (player.AuxVarGetFlt "_asOn") == 1
		let sOn := "on"
		let sOnBtn := "Turn auto-smoke off"
	endif
	let iGap := player.AuxVarGetFlt "_asGap"
	if iGap == -1
		let sWhen := "only when you have nicotine withdrawal"
	elseif iGap == -2
		let sWhen := "when you're Tired (Hardcore), if another smoke would still help"
	elseif iGap == 0
		let sWhen := "keep one lit (the next as soon as the last goes out)"
	elseif iGap == 1
		let sWhen := "1 minute after the last one goes out"
	elseif iGap == 3
		let sWhen := "3 minutes after the last one goes out"
	elseif iGap == 5
		let sWhen := "5 minutes after the last one goes out"
	elseif iGap == 10
		let sWhen := "10 minutes after the last one goes out"
	else
		let sWhen := "20 minutes after the last one goes out"
	endif
	let sPacks := "off"
	let sPacksBtn := "Open packs: turn on"
	if eval (player.AuxVarGetFlt "_asPacks") == 1
		let sPacks := "on"
		let sPacksBtn := "Open packs: turn off"
	endif
	let sNic := "off"
	let sNicBtn := "Nicotine alertness: turn on"
	if eval (player.AuxVarGetFlt "_asNic") == 1
		let sNic := "on"
		let sNicBtn := "Nicotine alertness: turn off"
	endif
	if eval (player.AuxVarGetFlt "_asNicOk") == 0
		let sNic := "unavailable (its script didn't load, see the console)"
	endif

	let sBody := sv_Construct "Auto-smoke: %z.%rLights up: %z.%r" sOn sWhen
	let sBody := sBody + (sv_Construct "Opens a pack or carton when out of loose smokes: %z.%r" sPacks)
	let sBody := sBody + (sv_Construct "Nicotine alertness (Hardcore): %z. Each smoke eases sleep deprivation a little, less if you've smoked in the last 6 hours. A smoke within an hour of bed costs you Well Rested, and withdrawal tires you faster.%r" sNic)
	let sBody := sBody + (sv_Construct "Carrying: %.0f cigarettes, %.0f cigarillos, %.0f cigars, %.0f packs, %.0f cartons." iCigs iRillos iCigars iPacks iCartons)
	let rCb := CompileScript "AutoSmoke/OpenBtn.txt"
	MessageBoxExAlt rCb "^Auto Smoke^%z|%z|Change when to light up|%z|%z|Smoke one now|Close" sBody sOnBtn sPacksBtn sNicBtn
end
