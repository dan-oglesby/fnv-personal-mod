; Auto Smoke: first-run defaults, once per character (aux variables on the player).
;   _asOn     1 = auto-smoke on
;   _asGap    minutes of play between one smoke going out and lighting the next
;             (0 = keep one lit; -1 = only when you have nicotine withdrawal; -2 = when tired)
;   _asPacks  1 = open a pack (or a carton) when you have no loose smokes
;   _asIdle   seconds of play since the last smoke went out
; Nicotine alertness (added 2026-11-14; set up once more for characters from before it):
;   _asNic    1 = on (Hardcore only)
;   _asSmokeT game hours of the last 4 smokes (index 0-3), _asSmokeN how many recorded
;   _asLastSmokeH, _asSlpPrev, _asNoRest, _asNoRestTicks: bedtime and withdrawal bookkeeping
int iK

begin Function {}
	if eval (player.AuxVarGetFlt "_asInit") == 0
		player.AuxVarSetFlt "_asOn" 1
		player.AuxVarSetFlt "_asGap" 5
		player.AuxVarSetFlt "_asPacks" 1
		player.AuxVarSetFlt "_asInit" 1
	endif
	if eval (player.AuxVarGetFlt "_asInit2") == 0
		player.AuxVarSetFlt "_asNic" 1
		let iK := 0
		while iK < 4
			player.AuxVarSetFlt "_asSmokeT" -100 iK
			let iK += 1
		loop
		player.AuxVarSetFlt "_asSmokeN" 0
		player.AuxVarSetFlt "_asLastSmokeH" -100
		player.AuxVarSetFlt "_asNoRest" 0
		player.AuxVarSetFlt "_asInit2" 1
	endif
end
