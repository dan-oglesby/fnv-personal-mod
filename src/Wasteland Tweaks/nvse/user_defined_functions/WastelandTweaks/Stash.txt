; Wasteland Tweaks: the FNV Cheat Terminal's "personal storage" counts as your inventory (2026-11-16).
; Moves up to iCount of rItem from those boxes to you and returns how many moved (0 if the cheat
; terminal isn't loaded). The boxes are persistent references in Doc Mitchell's house
; (FalloutNVCheatTerminal.esp): Aid 0x00480C, Ammo 0x00480D, Armor 0x00480E, Misc 0x00480F,
; Weapons 0x004810, General Items 1-3 0x00A3E9-0x00A3EB. Only what you put there yourself is in them,
; so this is a cheat you choose, not one you trip into. RemoveItemTarget (JIP) moves items silently
; and whole (condition, mods and ownership kept). Used by StashTick.txt and Auto Smoke (Smart Loot's
; auto-consume reads the boxes itself, in Pick.txt).
int iPlug
int iIdx
int iId
int iHave
int iTake
int iMoved
int iCount
ref rItem
ref rBox

begin Function {rItem, iCount}
	SetFunctionValue 0
	if eval (IsModLoaded "FalloutNVCheatTerminal.esp") == 0
		return
	endif
	if eval (IsFormValid rItem) == 0 || iCount < 1
		return
	endif
	let iPlug := GetModIndex "FalloutNVCheatTerminal.esp"
	let iMoved := 0
	let iIdx := 0
	while iIdx < 8 && iMoved < iCount
		; 18444-18448 = 0x480C-0x4810 (aid, ammo, armor, misc, weapons); 41961-41963 = 0xA3E9-0xA3EB
		if iIdx < 5
			let iId := 18444 + iIdx
		else
			let iId := 41956 + iIdx
		endif
		let iIdx += 1
		let rBox := BuildRef iPlug iId
		if eval (IsFormValid rBox) == 0
			continue
		endif
		let iHave := rBox.GetItemCount rItem
		if iHave < 1
			continue
		endif
		let iTake := iCount - iMoved
		if iTake > iHave
			let iTake := iHave
		endif
		rBox.RemoveItemTarget rItem player iTake 0	; 0 = ownership travels with it
		let iMoved += iTake
	loop
	SetFunctionValue iMoved
end
