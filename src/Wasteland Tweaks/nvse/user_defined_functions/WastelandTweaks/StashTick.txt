; Wasteland Tweaks: draws on the cheat terminal's personal storage (Stash.txt), once a second of
; play, in combat too (registered by ln_wasteland_tweaks.txt only when FalloutNVCheatTerminal.esp is
; loaded; never in menus, paused or during VATS playback).
; - Fission batteries (with Fission Power): wearing power armor with no spare, one comes out.
; - Every 5th second, the food feed (Feed.txt): food and drink past one of each go into storage,
;   and come back out as you use them.
; Ammo is never moved, either way (2026-11-18). This used to top up the ammo in your weapon from
; storage whenever you fell under two magazines, and the feed filed spare ammo away: together that
; was ammo without limit or weight. Ammo filed before then stays in Ammo Storage; take it out at the
; terminal if you want it.
; Off: player.AuxVarSetFlt "_twNoStash" 1
int iFeedT
ref rBatt
ref rSuit
ref rStash
ref rFeed

begin Function {}
	if player.GetDead
		return
	endif
	if IsInKillCam
		return
	endif
	if eval (player.AuxVarGetFlt "_twNoStash") == 1
		return
	endif
	let rStash := CompileScript "WastelandTweaks/Stash.txt"
	let iFeedT := player.AuxVarGetFlt "*_twFeedT"
	let iFeedT += 1
	if iFeedT >= 5
		let iFeedT := 0
	endif
	player.AuxVarSetFlt "*_twFeedT" iFeedT

	if eval IsModLoaded "FissionPower.esp"
		let rSuit := player.GetEquippedObject 2
		if eval IsFormValid rSuit
			if eval IsPowerArmor rSuit
				let rBatt := BuildRef 0 211910	; Fission Battery (FalloutNV.esm 0x033BC6)
				if eval (player.GetItemCount rBatt) == 0
					call rStash rBatt 1
				endif
			endif
		endif
	endif

	if iFeedT == 0
		let rFeed := CompileScript "WastelandTweaks/Feed.txt"
		if eval IsFormValid rFeed
			call rFeed
		endif
	endif
end
