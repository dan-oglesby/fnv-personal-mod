; Wasteland Tweaks: the cheat terminal's personal storage as a food feed (2026-11-16). Every 5 s of
; play, from StashTick.txt (so only with FalloutNVCheatTerminal.esp loaded). Food and drink (not
; medicine, nothing addictive, no quest items): one of each kind stays with you, the rest goes into
; Aid Storage. When you run out of a kind, one comes back from Aid Storage or General Items 1-3, so
; Smart Loot's auto-consume and the Pip-Boy always have it.
; No ammo (2026-11-18): this used to file spare ammo into Ammo Storage and hand it back as you shot,
; which made ammo unlimited and weightless. Ammo filed before then stays in Ammo Storage.
; Ownership travels with items (stolen stays stolen). Moves are listed first and made after, since
; moving while walking an inventory shifts it.
; Off: player.AuxVarSetFlt "_twNoFeed" 1 (the draws from storage in StashTick, Smart Loot and Auto
; Smoke stay on).
int iPlug
int iPass
int iNum
int iIdx
int iK
int iN
int iM
int iHave
int iBox
int iId
ref rItem
ref rBox
ref rAidBox
ref rStash
array_var aMove
array_var aCnt
array_var aEntry

begin Function {}
	if eval (player.AuxVarGetFlt "_twNoFeed") == 1
		return
	endif
	let iPlug := GetModIndex "FalloutNVCheatTerminal.esp"
	let rAidBox := BuildRef iPlug 18444	; Aid Storage 0x00480C
	if eval (IsFormValid rAidBox) == 0
		return
	endif
	let rStash := CompileScript "WastelandTweaks/Stash.txt"
	let iPass := player.AuxVarGetFlt "*_twFeedN"
	let iPass += 1
	player.AuxVarSetFlt "*_twFeedN" iPass

	; 1. into storage: food and drink past one of each
	let aMove := Ar_Construct "array"
	let aCnt := Ar_Construct "array"
	let iNum := player.GetNumItems
	let iIdx := 0
	while iIdx < iNum
		let rItem := player.GetInventoryObject iIdx
		let iIdx += 1
		if eval (IsFormValid rItem) == 0
			continue
		endif
		if eval (GetType rItem) != 47
			continue
		endif
		if eval IsQuestItem rItem
			continue
		endif
		if eval (GetIngestibleFlag rItem 2) == 0 || (GetIngestibleFlag rItem 4) || (GetAddictionChance rItem) > 0
			continue
		endif
		let iHave := player.GetItemCount rItem
		if iHave > 1
			let iM := iHave - 1
			Ar_Append aMove rItem
			Ar_Append aCnt iM
		endif
	loop
	let iN := Ar_Size aMove
	let iK := 0
	while iK < iN
		let rItem := aMove[iK]
		let iM := aCnt[iK]
		let iK += 1
		player.RemoveItemTarget rItem rAidBox iM 0
	loop

	; 2. back from storage: one of each food or drink you've run out of
	let aMove := Ar_Construct "array"
	let iBox := 0
	while iBox < 4
		; Aid Storage, then General Items 1-3 (41961-41963 = 0xA3E9-0xA3EB)
		let iId := 18444
		if iBox > 0
			let iId := 41960 + iBox
		endif
		let iBox += 1
		let rBox := BuildRef iPlug iId
		if eval (IsFormValid rBox) == 0
			continue
		endif
		let iNum := rBox.GetNumItems
		let iIdx := 0
		while iIdx < iNum
			let rItem := rBox.GetInventoryObject iIdx
			let iIdx += 1
			if eval (IsFormValid rItem) == 0
				continue
			endif
			if eval (GetType rItem) != 47
				continue
			endif
			if eval (GetIngestibleFlag rItem 2) == 0 || (GetIngestibleFlag rItem 4) || (GetAddictionChance rItem) > 0
				continue
			endif
			if eval IsQuestItem rItem
				continue
			endif
			if eval (player.GetItemCount rItem) == 0 && (AuxVarGetFlt "*_twFedN" 0 rItem) != iPass
				AuxVarSetFlt "*_twFedN" iPass 0 rItem
				Ar_Append aMove rItem
			endif
		loop
	loop
	foreach aEntry <- aMove
		let rItem := aEntry["value"]
		call rStash rItem 1
	loop
	let aMove := Ar_Null
	let aCnt := Ar_Null
	let aEntry := Ar_Null
end
