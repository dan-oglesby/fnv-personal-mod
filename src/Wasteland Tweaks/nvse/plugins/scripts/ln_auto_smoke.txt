; Auto Smoke (2026-11-11; part of Wasteland Tweaks since 2026-11-15): an add-on for "Light Up and Smoke Those Cigarettes" (edisleado,
; Nexus 49677). Lights your next smoke for you, the same as using it from the Pip-Boy, so the
; mod's own lighting, effects, addiction and butts all apply. Butts and empty packs and cartons are
; dropped where you stand (Litter.txt), like a real smoker.
;
; Settings: equip the Cigarette Case (Auto Smoke) from the Pip-Boy's Apparel tab; the menu opens
; when you close the Pip-Boy, the same way as the Wasteland Bank Card. The case is given on the
; first load and again if it is ever lost. It lives in AutoSmoke.esp (one item, no scripts; made
; by make_plugins.py). From the console: call (CompileScript "AutoSmoke/Open.txt")
;
; Run by JIP LN NVSE's script runner (ln_: new game and every load). Does nothing unless
; LightUpAndSmokeThoseCigarettes_edisleado.esp is loaded. Settings are JIP auxiliary variables on
; the player ("_as..."), kept in JIP's co-save. Needs xNVSE 6.2.8+, JIP LN 56.95+.
;
; Re-registering here on every load is safe: xNVSE clears repeating calls on load and new game,
; and the equip handler is removed before it is added again.

if eval IsModLoaded "LightUpAndSmokeThoseCigarettes_edisleado.esp"
	; Every function must have compiled. xNVSE compiles them at startup; a failed one comes back
	; empty, and a timer or event handler that calls an empty function can crash xNVSE (2026-11-13:
	; two crashes in nvse_1_4.dll that way, from a variable named like a console command). So if any
	; failed, register nothing and say so.
	let int iBad := 0
	let ref rChk := CompileScript "AutoSmoke/Always.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/Always.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/Init.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/Init.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/IsLit.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/IsLit.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/LightUp.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/LightUp.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/OnEquip.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/OnEquip.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/OnKey.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/OnKey.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/Open.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/Open.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/OpenBtn.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/OpenBtn.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/Tick.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/Tick.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/UseCase.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/UseCase.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/Litter.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/Litter.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/LitKind.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/LitKind.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/NicCount.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/NicCount.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/NicSmoke.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/NicSmoke.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/NicTick.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/NicTick.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "AutoSmoke/NicMenu.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Wasteland Tweaks: AutoSmoke/NicMenu.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	; NicSleep is the only script that names the sleep-deprivation actor value. It is optional: if it
	; failed to compile, only nicotine alertness switches off (_asNicOk 0), not Auto Smoke.
	let rChk := CompileScript "AutoSmoke/NicSleep.txt"
	if eval (IsFormValid rChk) == 0
		player.AuxVarSetFlt "_asNicOk" 0
		print "Auto Smoke: nicotine alertness is off because NicSleep.txt failed to compile (see nvse.log)"
	else
		player.AuxVarSetFlt "_asNicOk" 1
	endif
	if iBad > 0
		print "Wasteland Tweaks: " + $iBad + " script function(s) failed to compile (named above). Everything else in it is running."
		MessageEx "Wasteland Tweaks: part of it is off, a script failed to compile (see the console)"
	endif
	if iBad >= 0	; always (2026-11-17): each part below is registered only if its own function compiled
		let ref rInit := CompileScript "AutoSmoke/Init.txt"
		if eval (IsFormValid rInit)
			call rInit
		endif
		if eval IsModLoaded "AutoSmoke.esp"
			let int iAS := GetModIndex "AutoSmoke.esp"
			let ref rCase := BuildRef iAS 2048	; ASCigaretteCase 0x000800
			if eval (player.GetItemCount rCase) == 0
				player.AddItem rCase 1
			endif
			let ref rOnEquip := CompileScript "AutoSmoke/OnEquip.txt"
			if eval (IsFormValid rOnEquip)
				RemoveEventHandler "OnActorEquip" rOnEquip
			endif
			; Filtered to the cigarette case ("second" = the equipped item): xNVSE queues equip events that fire on a
			; background thread onto an unlocked list the main thread clears every frame, and an unfiltered
			; handler sends every NPC equip during cell loads there. That race corrupted memory and crashed
			; the game's loaders (2026-11-14). A filtered handler never sees other items.
			if eval (IsFormValid rOnEquip)
				SetEventHandler "OnActorEquip" rOnEquip "second"::rCase
			endif
		else
			print "Auto Smoke: AutoSmoke.esp is not enabled, so there is no cigarette case to open the settings with"
		endif
		; Optional hotkey as well (L = DirectInput 38): remove the ; from the next two lines.
		;let ref rKey := CompileScript "AutoSmoke/OnKey.txt"
		;SetOnKeyDownEventHandler rKey 1 38
		; nicotine alertness: notice the Sleep/Wait menu (1012) for the bedtime rule (JIP keeps
		; handlers in a set, so re-adding on every load doesn't stack), and record what is already lit
		; so a smoke carried over a load isn't counted as a new one
		let ref rMenu := CompileScript "AutoSmoke/NicMenu.txt"
		if eval (IsFormValid rMenu)
			SetOnMenuOpenEventHandler rMenu 1 1012
		endif
		let ref rKind := CompileScript "AutoSmoke/LitKind.txt"
		let int iKindNow := call rKind
		player.AuxVarSetFlt "_asWasLit" iKindNow
		; check every 5 s of play: flags 4 + 16 = not in menus, not while paused
		let ref rTick := CompileScript "AutoSmoke/Tick.txt"
		let ref rTrue := CompileScript "AutoSmoke/Always.txt"
		if eval (IsFormValid rTick) && (IsFormValid rTrue)
			CallWhilePerSeconds 5 rTick rTrue 20
		endif
	endif
endif
