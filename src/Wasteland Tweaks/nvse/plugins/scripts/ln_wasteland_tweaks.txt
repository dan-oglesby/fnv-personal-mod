; Wasteland Tweaks (Dan, 2026-11-15): small adjustments that don't need a mod of their own. No
; plugin. Run by JIP LN NVSE's script runner (ln_: new game and every load); game settings are not
; saved, so they are set again on every load. Needs xNVSE.
;
; 1. Lasting shells: spent casings stay on the ground for a long time, the way Permanent Weapon
;    Shells did, but with the two numbers apart: how long a casing lasts (fGunShellLifetime, real
;    seconds) and how many can exist at once (iDebrisMaxCount; past it the game removes the oldest
;    itself). Defaults: 1 hour, 1,500 casings. Permanent Weapon Shells "Very Heavy" (a day, 10,000)
;    plus ITR's SmallDebrisCap (keep it off: bSmallDebrisCap=0) were behind two casing crashes on
;    2026-11-14. Change them from the console, then reload:
;      player.AuxVarSetFlt "_twShellLife" 7200     seconds a casing lasts (0 = leave the game's own)
;      player.AuxVarSetFlt "_twShellMax" 2500      casings at once (0 = leave the game's own)
;    Don't run Permanent Weapon Shells alongside: it sets the same two settings.
;
; 2. VATS rescue: VATS playback sometimes hangs (seen 2026-11-15 when the Mysterious Stranger
;    appeared). A watchdog ends the VATS camera if it runs 15 s without a break, and the End key ends
;    it any time. Smart Loot, Auto Smoke, Fission Power and Wasteland Market also stop using items
;    during VATS playback. Settings (console):
;      player.AuxVarSetFlt "_twVatsMax" 20        seconds before the watchdog ends it (default 15)
;      player.AuxVarSetFlt "_twVatsKey" 199       rescue key, DirectInput code (default 207, End;
;                                                 0 = no key)
;
; 3. Console log: everything the console shows also goes to ConsoleLog.txt in the game folder, from
;    the start of each game launch: our mods' prints, any other mod's print or printc messages, and
;    the game's own console output (vanilla SetConsoleOutputFile, which xNVSE offers scripts as
;    con_SCOF). The session watcher keeps a copy per session. To stop it from the next launch:
;      player.AuxVarSetFlt "_twNoConsoleLog" 1
;    Other mods' scripts can't be made to log what they do step by step without editing them; this
;    catches whatever they print, and falloutnv_error.log already records their script errors.
;
; 4. Cheat terminal storage (only with FalloutNVCheatTerminal.esp): its "personal storage" boxes
;    count as your inventory, and work as a feed: ammo and food over a working amount go into
;    them (four magazines for the ammo in your hands, one for each kind your other weapons take,
;    one of each food and drink) and come back out as you use them. A fission battery comes out
;    when you wear power armor with no spare (Fission Power), Smart Loot's auto-consume uses aid
;    from them when you carry none, and Auto Smoke lights smokes from them. Nothing is in those
;    boxes unless you put it there. To turn it all off, or just the feed into storage:
;      player.AuxVarSetFlt "_twNoStash" 1
;      player.AuxVarSetFlt "_twNoFeed" 1

if eval (player.AuxVarGetFlt "_twInit") < 1
	player.AuxVarSetFlt "_twInit" 1
	player.AuxVarSetFlt "_twShellLife" 3600
	player.AuxVarSetFlt "_twShellMax" 1500
endif
let float fTwLife := player.AuxVarGetFlt "_twShellLife"
let float fTwMax := player.AuxVarGetFlt "_twShellMax"
if fTwLife > 0
	SetNumericGameSetting "fGunShellLifetime" fTwLife
endif
if fTwMax > 0
	SetNumericGameSetting "iDebrisMaxCount" fTwMax
endif

; VATS rescue (needs JIP LN and JohnnyGuitar NVSE; switched off if its scripts fail to compile)
if eval (player.AuxVarGetFlt "_twInit") < 2
	player.AuxVarSetFlt "_twInit" 2
	player.AuxVarSetFlt "_twVatsMax" 15
	player.AuxVarSetFlt "_twVatsKey" 207
endif
let ref rTwWatch := CompileScript "WastelandTweaks/VatsWatch.txt"
let ref rTwKey := CompileScript "WastelandTweaks/VatsKey.txt"
let ref rTwTrue := CompileScript "WastelandTweaks/Always.txt"
if eval (IsFormValid rTwWatch) && (IsFormValid rTwKey) && (IsFormValid rTwTrue)
	; once a second, outside menus (VATS playback isn't a menu) and not while paused: flags 4 + 16
	CallWhilePerSeconds 1 rTwWatch rTwTrue 20
	let int iTwKey := player.AuxVarGetFlt "_twVatsKey"
	if iTwKey > 0
		SetOnKeyDownEventHandler rTwKey 1 iTwKey
	endif
else
	print "Wasteland Tweaks: the VATS rescue is off because its scripts failed to compile (see nvse.log)"
endif

; console log, once per game launch (GetGameRestarted is true only the first time after a restart,
; so loading a save doesn't start the file over)
if eval (player.AuxVarGetFlt "_twNoConsoleLog") == 0
	if GetGameRestarted
		con_SCOF "ConsoleLog.txt"
	endif
endif

; cheat terminal storage link, once a second outside menus (flags 4 + 16), only with the terminal
if eval IsModLoaded "FalloutNVCheatTerminal.esp"
	let ref rTwStash := CompileScript "WastelandTweaks/StashTick.txt"
	let ref rTwStashFn := CompileScript "WastelandTweaks/Stash.txt"
	if eval (IsFormValid rTwStash) && (IsFormValid rTwStashFn) && (IsFormValid rTwTrue)
		CallWhilePerSeconds 1 rTwStash rTwTrue 20
	else
		print "Wasteland Tweaks: the cheat terminal storage link is off because its scripts failed to compile (see nvse.log)"
	endif
endif
