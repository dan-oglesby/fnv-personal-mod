; Fission Power (2026-11-15): power armor runs on fission batteries. Walking drains the one in
; the suit, and so, more slowly, does just having it online; when it runs out the spent battery is ejected at your feet and the next goes in. With none
; left the suit is dead weight (Agility -2, no Strength bonus, 25 percent slower) until you find
; another. Batteries are
; the only fuel. Settings (console):
;   player.AuxVarSetFlt "_fpLife" 20000    steps per battery (default 10000, about 7.5 km)
;   player.AuxVarSetFlt "_fpPen" 0         no dead-suit penalty (1 = on)
;   player.AuxVarSetFlt "_fpSlow" 40       a dead suit slows you 40 percent (default 25)
;   player.AuxVarSetFlt "_fpIdle" 2        standby draw, steps per second worn (default 1, 0 = none)
;   player.AuxVarSetFlt "_fpSnd" 1         eject sound: battle rifle reload part 1, 2 or 3 (default 3)
;   player.AuxVarSetFlt "_fpFT" 0          fast travel is free (default 1: it drains by distance)
; Shows on the HUD through Needs Meters (PWR). Smart Loot keeps a few batteries while you own power
; armor. FissionPower.esp holds the Depleted Fission Battery (and the old Nutrient Paste, no longer made).
;
; The suit's food and water (2026-11-19, Suit.txt; it replaces 2026-11-17's Nutrient Paste): two
; reservoirs in need points, shown by Needs Meters after the H2O and FOD numbers ("H2O 3 +180" is 3
; thirst and 180 in the suit). In Hardcore the suit takes the food and drink you carry into them, an
; item every 3 seconds (SuitTake.txt): an item that gives both fills both, and only when all of it
; fits, so none is wasted; irradiated items are filtered clean while it has power and stay in your
; pack while it has none. It feeds you from them, exactly as much as thirst and hunger build up, so
; both stay at 0 while it holds any, with or without power. It keeps only water and food (healing
; and short buffs don't carry over) and leaves you quest items, anything addictive, and items that
; also ease sleep or radiation or run a script (a Sunset Sarsaparilla's cap). With Bottle That
; Water, a bottle it takes leaves its empty bottle.
;   player.AuxVarSetFlt "_fpWaterMax" 600   water reservoir size, thirst points (default 400)
;   player.AuxVarSetFlt "_fpFoodMax" 600    food reservoir size, hunger points (default 400)
;   player.AuxVarSetFlt "_fpFilter" 4       battery steps per point to filter irradiated food or water (default 2)
;   player.AuxVarSetFlt "_fpTake" 0         the suit takes nothing in (default 1)
;   player.AuxVarSetFlt "_fpFeed" 0         the suit doesn't feed you (default 1)
; Standing in water or swimming fills the water tank from the water itself (2026-11-19), faster the
; deeper you are; irradiated water is filtered at "_fpFilter" a point, so it needs power.
;   player.AuxVarSetFlt "_fpDraw" 0         the suit doesn't draw water (default 1)
; With no fission batteries left, energy weapon ammo powers the suit (2026-11-17, Cells.txt): each
; cell worth its vanilla value against a battery's 75 caps (an energy cell about 267 steps), least
; energy-dense first, two magazines of your weapon's loaded ammo kept back, spent cells returned
; drained. player.AuxVarSetFlt "_fpAmmo" 0 = batteries only. By default each cell runs on its own,
; with a click and a count when it's spent; player.AuxVarSetFlt "_fpCellOne" 0 refuels a fifth of a
; battery at a time instead. Drained cells go into your pack. Smart Loot keeps energy ammo as fuel.
; The worn suit weighs more by its processor ("_fpProc", 5 lb) and what its reservoirs hold
; (SuitWeight.txt).
;
; Run by JIP LN NVSE's script runner (ln_: new game and every load). Needs xNVSE 6.4+ and JIP LN.

if eval IsModLoaded "FissionPower.esp"
	; Every function must have compiled: a timer that calls a function that failed to compile can
	; crash xNVSE (2026-11-13). If one failed, register nothing and say so.
	let int iBad := 0
	let ref rChk := CompileScript "FissionPower/Always.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/Always.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/Init.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/Init.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/Penalty.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/Penalty.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/Tick.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/Tick.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/Cells.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/Cells.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/Suit.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/Suit.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/SuitTake.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/SuitTake.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	let rChk := CompileScript "FissionPower/SuitWeight.txt"
	if eval (IsFormValid rChk) == 0
		let iBad += 1
		print "Fission Power: FissionPower/SuitWeight.txt failed to compile, so the part that uses it is off (details: nvse.log)"
	endif
	if iBad > 0
		print "Fission Power: " + $iBad + " script function(s) failed to compile (named above). Everything else in it is running."
		MessageEx "Fission Power: part of it is off, a script failed to compile (see the console)"
	endif
	if iBad >= 0	; always (2026-11-17): each part below is registered only if its own function compiled
		let ref rInit := CompileScript "FissionPower/Init.txt"
		if eval (IsFormValid rInit)
			call rInit
		endif
		player.AuxVarSetFlt "*_fpHave" 0
		; once a second of play: flags 4 + 16 = not in menus, not while paused
		let ref rTick := CompileScript "FissionPower/Tick.txt"
		let ref rTrue := CompileScript "FissionPower/Always.txt"
		let ref rStart := CompileScript "FissionPower/Start.txt"
		if eval (IsFormValid rTick) && (IsFormValid rTrue) && (IsFormValid rStart)
			CallAfterSeconds 0.075 rStart 1 rTick rTrue 1 20	; timer slot 0.075 s (tools/timer_slots.py)
		endif
	endif
endif
