; Fission Power: the suit's food and water (2026-11-19; it replaces the Nutrient Paste of 2026-11-17).
; Once a second while you wear power armor (Tick.txt), with or without power. The suit has two
; reservoirs, counted in the game's own need points: water ("_fpWater", up to "_fpWaterMax", default
; 400) and food ("_fpFood", up to "_fpFoodMax", default 400). Needs Meters shows them after the H2O
; and FOD numbers: "H2O 3 +180" is 3 points of thirst and 180 in the suit.
; In Hardcore only (outside it food and water do nothing for needs, so the suit leaves them be):
; 1. Taking in, every 3 seconds: the food and drink you carry go into the reservoirs, one item at a
;    time (SuitTake.txt picks it). An item that gives both fills both, and it is only taken when both
;    have room for all of it, so none of it is wasted. Salty food's thirst comes out of the water
;    reservoir. Irradiated items are filtered clean while the suit has power ("_fpFilter" battery
;    steps per point, default 2) and stay in your pack while it has none. With Bottle That Water, a
;    bottle of water leaves its empty bottle behind, as drinking it does. The suit keeps only water
;    and food: an item's healing and short buffs don't carry over. Quest items, anything addictive,
;    and items that also ease sleep or radiation or run a script (a Sunset Sarsaparilla's bottle cap)
;    stay in your pack. Off: player.AuxVarSetFlt "_fpTake" 0.
; 2. Feeding you, every second: when thirst or hunger has built up a point or more, the suit gives
;    you exactly that much from its reservoir, so both stay at 0 while it holds any. It lowers the
;    need's base value, which is what the game's own thirst and hunger raise (see Mojave Heat's
;    Tick.txt). Off: player.AuxVarSetFlt "_fpFeed" 0.
; The reservoirs stay as they are while you're out of power armor, and carry on when you put one on.
int iT
int iPow
int iPlug
float fWat
float fFod
float fWMax
float fFMax
float fRoomW
float fRoomF
float fGotW
float fGotF
float fNeed
float fBase
float fGive
float fCost
float fLeft
ref rItem
ref rFn
ref rBottle

begin Function {}
	let fWat := player.AuxVarGetFlt "_fpWater"
	let fFod := player.AuxVarGetFlt "_fpFood"
	let fWMax := player.AuxVarGetFlt "_fpWaterMax"
	if fWMax <= 0
		let fWMax := 400
	endif
	let fFMax := player.AuxVarGetFlt "_fpFoodMax"
	if fFMax <= 0
		let fFMax := 400
	endif

	if eval player.IsHardcore
		; 1. taking in, every 3 seconds. The timer is saved before the search, so a search that fails
		; can't make it run (and fail) every second (the paste's mulching search did, 2026-11-18).
		let iT := player.AuxVarGetFlt "*_fpTakeT"
		let iT += 1
		if iT >= 3
			let iT := 0
		endif
		player.AuxVarSetFlt "*_fpTakeT" iT
		if eval iT == 0 && (player.AuxVarGetFlt "_fpTake") != 0
			let fRoomW := fWMax - fWat
			let fRoomF := fFMax - fFod
			let iPow := 0
			if eval (player.AuxVarGetFlt "_fpLeft") > 0
				let iPow := 1
			endif
			let rItem := 0
			let rFn := CompileScript "FissionPower/SuitTake.txt"
			if eval IsFormValid rFn
				let rItem := call rFn fRoomW fRoomF fWat iPow
			endif
			if eval IsFormValid rItem
				let fGotW := player.AuxVarGetFlt "*_fpTakeW"
				let fGotF := player.AuxVarGetFlt "*_fpTakeF"
				player.RemoveItem rItem 1 1
				let fWat += fGotW
				let fFod += fGotF
				if fWat < 0
					let fWat := 0
				endif
				player.AuxVarSetFlt "_fpWater" fWat
				player.AuxVarSetFlt "_fpFood" fFod
				; filtering an irradiated item costs power
				if eval (player.AuxVarGetFlt "*_fpTakeRad") == 1
					let fCost := player.AuxVarGetFlt "_fpFilter"
					let fLeft := player.AuxVarGetFlt "_fpLeft"
					let fLeft -= (player.AuxVarGetFlt "*_fpTakeAll") * fCost
					if fLeft < 1
						let fLeft := 1
					endif
					player.AuxVarSetFlt "_fpLeft" fLeft
				endif
				; Bottle That Water: the empty bottle stays with you (LFoxWaterBottleEmpty 0xADE)
				if eval (player.AuxVarGetFlt "*_fpTakeBtl") == 1
					let iPlug := GetModIndex "LFox Bottle That Water.esp"
					let rBottle := BuildRef iPlug 2782
					if eval IsFormValid rBottle
						player.AddItem rBottle 1 1
					endif
				endif
				if eval (player.AuxVarGetFlt "_fpTakeSeen") == 0
					player.AuxVarSetFlt "_fpTakeSeen" 1
					MessageBoxEx "Power armor%rThe suit takes the food and water you carry into its reservoirs and feeds you from them, keeping thirst and hunger at 0. An item that gives both fills both, and nothing goes in that doesn't fit whole. Needs Meters shows what each reservoir holds after the H2O and FOD numbers."
				endif
			endif
		endif

		; 2. feeding you, every second: whatever thirst and hunger have built up, exactly, once it's a
		; point or more. A need is held in its base value (the game's own thirst and hunger raise it,
		; and Dead Money's Auto-Doc clears sleep with SetAV), so that's what comes down; if the base
		; somehow holds less than the need, RestoreAV does it (as Bottle That Water's canteen and the
		; cheat terminal do). The reservoir pays only for what actually came off.
		if eval (player.AuxVarGetFlt "_fpFeed") != 0
			let fNeed := player.GetAV Dehydration
			if fNeed >= 1 && fWat > 0
				let fGive := fNeed
				if fGive > fWat
					let fGive := fWat
				endif
				let fBase := player.GetBaseAV Dehydration
				if fBase >= fGive
					let fBase -= fGive
					player.SetAV Dehydration fBase
				else
					player.RestoreAV Dehydration fGive
				endif
				let fGive := fNeed - (player.GetAV Dehydration)
				if fGive > 0
					let fWat -= fGive
					if fWat < 0
						let fWat := 0
					endif
				endif
			endif
			let fNeed := player.GetAV Hunger
			if fNeed >= 1 && fFod > 0
				let fGive := fNeed
				if fGive > fFod
					let fGive := fFod
				endif
				let fBase := player.GetBaseAV Hunger
				if fBase >= fGive
					let fBase -= fGive
					player.SetAV Hunger fBase
				else
					player.RestoreAV Hunger fGive
				endif
				let fGive := fNeed - (player.GetAV Hunger)
				if fGive > 0
					let fFod -= fGive
					if fFod < 0
						let fFod := 0
					endif
				endif
			endif
			player.AuxVarSetFlt "_fpWater" fWat
			player.AuxVarSetFlt "_fpFood" fFod
		endif
	endif

	; the HUD (Needs Meters): what each reservoir holds
	SetUIFloat "HUDMainMenu/_NMSuitW" fWat
	SetUIFloat "HUDMainMenu/_NMSuitF" fFod
end
