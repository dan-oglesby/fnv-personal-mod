; Fission Power: the suit's food system (2026-11-17), once a second while you wear powered armor
; (Tick.txt). The suit keeps a reservoir of Nutrient Paste ("_fpPaste", in hunger points; it holds
; "_fpPasteMax", default 400; one serving, FissionPower.esp FPNutrientPaste 0x801, is 40).
; 1. Paste you carry goes into the reservoir while there's room (no power).
; 2. Every 10 seconds, if there's room for a whole food item's nutrition (Nutri.txt), the suit
;    mulches one: the heaviest for what it gives first, so the most weight leaves your pack. Richer
;    food makes more paste. It costs power: "_fpMulch" steps' worth of battery per hunger point
;    (default 1: Cram, 50 hunger, about 50 seconds of standby).
; 3. In Hardcore, out of combat, when hunger comes within the margin of its first penalty stage
;    (Smart Loot's "_seEarly", or 50), the suit feeds you a serving from the reservoir. Off with
;    player.AuxVarSetFlt "_fpFeed" 0 (and mulching with "_fpMulch" 0).
int iPlug
int iN
int iRoom
int iT
float fRes
float fMax
float fRoom
float fBestMag
float fCost
float fLeft
float fAt
float fNeed
ref rPaste
ref rBest
ref rFn
ref rStage

begin Function {}
	let iPlug := GetModIndex "FissionPower.esp"
	let rPaste := BuildRef iPlug 2049	; FPNutrientPaste
	if eval (IsFormValid rPaste) == 0
		return
	endif
	let fRes := player.AuxVarGetFlt "_fpPaste"
	let fMax := player.AuxVarGetFlt "_fpPasteMax"
	if fMax <= 0
		let fMax := 400
	endif

	; 1. carried paste into the reservoir
	let iN := player.GetItemCount rPaste
	if iN > 0
		let iRoom := Floor ((fMax - fRes) / 40)
		if iN > iRoom
			let iN := iRoom
		endif
		if iN > 0
			player.RemoveItem rPaste iN 1
			let fRes += iN * 40
		endif
	endif

	; save what step 1 did before anything that could fail (2026-11-18: the mulching search used to
	; fail on every run, which skipped the saves at the end, so paste moved in was lost)
	player.AuxVarSetFlt "_fpPaste" fRes

	; 2. mulching, every 10 seconds. The timer is saved before the search (Mulch.txt) runs, so a
	; failing search can't make it run, and fail, every second.
	let fCost := player.AuxVarGetFlt "_fpMulch"
	let iT := player.AuxVarGetFlt "*_fpMulchT"
	let iT += 1
	if eval iT >= 10 && fCost > 0
		let iT := 0
		player.AuxVarSetFlt "*_fpMulchT" 0
		let rBest := 0
		let fBestMag := 0
		let rFn := CompileScript "FissionPower/Mulch.txt"
		if eval IsFormValid rFn
			let fRoom := fMax - fRes
			let rBest := call rFn fRoom
			let fBestMag := player.AuxVarGetFlt "*_fpMulchMag"
		endif
		if eval (IsFormValid rBest) && fBestMag > 0
			player.RemoveItem rBest 1 1
			let fRes += fBestMag
			let fLeft := player.AuxVarGetFlt "_fpLeft"
			let fLeft -= fBestMag * fCost
			if fLeft < 1
				let fLeft := 1
			endif
			player.AuxVarSetFlt "_fpLeft" fLeft
			if eval (player.AuxVarGetFlt "_fpMulchSeen") == 0
				player.AuxVarSetFlt "_fpMulchSeen" 1
				MessageBoxEx "Power armor%rThe suit's processor mulches the food you carry into Nutrient Paste whenever its reservoir has room, at a little power, and feeds you from it in Hardcore. Paste made at a campfire costs no power."
			endif
		endif
	endif
	player.AuxVarSetFlt "*_fpMulchT" iT

	; 3. feeding
	if eval fRes >= 40 && (player.AuxVarGetFlt "_fpFeed") == 1
		if eval player.IsHardcore
			if eval (player.IsInCombat) == 0
				let rStage := BuildRef 0 1448198	; hunger1 (0x161906)
				let fAt := GetHardcoreStageThreshold rStage
				if fAt <= 0
					let fAt := 200
				endif
				let fNeed := player.AuxVarGetFlt "_seEarly"
				if fNeed <= 0
					let fNeed := 50
				endif
				if eval (player.GetAV Hunger) >= fAt - fNeed
					player.AddItem rPaste 1 1
					player.EquipItem rPaste 0 1
					let fRes -= 40
				endif
			endif
		endif
	endif
	player.AuxVarSetFlt "_fpPaste" fRes
end
