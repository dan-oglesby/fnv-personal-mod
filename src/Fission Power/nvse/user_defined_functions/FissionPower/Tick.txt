; Fission Power: once a second of play (registered by ln_fission_power.txt; not in menus or paused).
; While you wear power armor (upper body slot, flagged as power armor; NCR Salvaged Power Armor has
; no servos and isn't), the distance you cover drains its battery: one step = 52.5 game units (0.75 m),
; one battery = "_fpLife" steps (default 10,000, about 7.5 km). Just having the suit online also
; draws power: "_fpIdle" steps' worth per second of play (default 1), so a battery lasts about 2.8
; hours standing still, against about half an hour running. Fast travel drains by the distance
; covered (x1.25, roads wind; setting "_fpFT"). Doors don't count: a change of cell (or worldspace,
; outside), or a jump of more than 1,000 units that isn't within one outdoor worldspace, is skipped.
; When the battery runs out the spent one is ejected (tossed over your left shoulder as a Depleted
; Fission Battery, Eject.txt, with the battle rifle's clip-spring sound) and a fresh one from your
; inventory goes in. With none left, the suit is dead weight
; (Penalty.txt) until you pick one up. Needs Meters shows the charge (PWR) and your spares.
int iPlug
int iSpare
int iUsed
int iFT
int iSnd
float fX
float fY
float fDist
float fLife
float fLeft
float fPct
ref rSuit
ref rWhere
ref rLast
ref rBatt
ref rSpent
ref rFn
ref rSnd
ref rCell
ref rEject
float fNeed
float fGot
int iCells

begin Function {}
	if player.GetDead
		return
	endif
	; nothing during VATS playback (JIP IsInKillCam): using or equipping items mid-VATS can leave the
	; VATS camera stuck (2026-11-15, a Mysterious Stranger appearance froze while a radscorpion closed in)
	if IsInKillCam
		return
	endif
	let rFn := CompileScript "FissionPower/Penalty.txt"
	let rSuit := player.GetEquippedObject 2
	if eval (IsFormValid rSuit) == 0 || (IsPowerArmor rSuit) == 0
		SetUIFloat "HUDMainMenu/_NMPower" -1
		player.AuxVarSetFlt "*_fpHave" 0
		let rCell := CompileScript "FissionPower/SuitWeight.txt"
		call rCell rSuit 0	; a suit we made heavier gets its weight back
		call rFn 0 rSuit
		; no suit: the HUD's reservoir numbers go (what the reservoirs hold waits there, Suit.txt)
		SetUIFloat "HUDMainMenu/_NMSuitW" -1
		SetUIFloat "HUDMainMenu/_NMSuitF" -1
		return
	endif

	; distance since the last second, in the same place
	let fX := player.GetPos X
	let fY := player.GetPos Y
	let rWhere := player.GetParentWorldspace
	if eval (IsFormValid rWhere) == 0
		let rWhere := player.GetParentCell
	endif
	let fDist := 0
	if eval (player.AuxVarGetFlt "*_fpHave") == 1
		let rLast := player.AuxVarGetRef "*_fpWhere"
		if eval rLast == rWhere
			let fDist := ((fX - (player.AuxVarGetFlt "*_fpX")) ^ 2 + (fY - (player.AuxVarGetFlt "*_fpY")) ^ 2) ^ 0.5
			if fDist > 1000
				; a jump: fast travel if it stayed in the same outdoor worldspace, otherwise (a door,
				; an interior move) not counted. Fast travel drains by the distance covered, times
				; 1.25 because roads wind ("_fpFT" 1, the default; 0 = fast travel is free)
				let fDist := 0
				if eval (IsFormValid (player.GetParentWorldspace)) && (player.AuxVarGetFlt "_fpFT") == 1
					let fDist := ((fX - (player.AuxVarGetFlt "*_fpX")) ^ 2 + (fY - (player.AuxVarGetFlt "*_fpY")) ^ 2) ^ 0.5
					let fDist *= 1.25
					let iFT := 1
				endif
			endif
		endif
	endif
	player.AuxVarSetFlt "*_fpX" fX
	player.AuxVarSetFlt "*_fpY" fY
	player.AuxVarSetRef "*_fpWhere" rWhere
	player.AuxVarSetFlt "*_fpHave" 1

	let fLife := player.AuxVarGetFlt "_fpLife"
	if fLife < 100
		let fLife := 10000
	endif
	let fLeft := player.AuxVarGetFlt "_fpLeft"
	let fLeft -= fDist / 52.5
	if fLeft > 0
		let fLeft -= player.AuxVarGetFlt "_fpIdle"	; standby draw, once a second
	endif
	let rBatt := BuildRef 0 211910	; Fission Battery (FalloutNV.esm 0x033BC6)
	let iSpare := player.GetItemCount rBatt
	if fLeft <= 0
		; eject spent batteries and load fresh ones until the charge is back above 0 (a long fast
		; travel can use several), or until there are none left
		let iUsed := 0
		let iPlug := GetModIndex "FissionPower.esp"
		let rSpent := BuildRef iPlug 2048	; Depleted Fission Battery (FPDepletedBattery 0x000800)
		; tossed over your left shoulder (Eject.txt); dropped at your feet if it failed to compile
		let rEject := CompileScript "FissionPower/Eject.txt"
		while fLeft <= 0 && iSpare > 0
			player.RemoveItem rBatt 1 1
			if eval IsFormValid rEject
				call rEject rSpent
			else
				player.PlaceAtMe rSpent 1
			endif
			let fLeft += fLife
			let iSpare -= 1
			let iUsed += 1
		loop
		; then energy weapon ammo (Cells.txt), least energy-dense first, refuelling to a fifth of a
		; battery at a time; spent cells come back as drained cells ("_fpAmmo" 0 = batteries only)
		if eval fLeft <= 0 && (player.AuxVarGetFlt "_fpAmmo") == 1
			let rCell := CompileScript "FissionPower/Cells.txt"
			; one cell at a time, each spent one popped out with a click and a count ("_fpCellOne" 1, the
			; default), or a fifth of a battery at once (0); drained cells go into your pack either way
			let fNeed := fLife * 0.2 - fLeft
			if eval (player.AuxVarGetFlt "_fpCellOne") == 1
				let fNeed := 1 - fLeft
			endif
			let fGot := call rCell fNeed 1
			if fGot > 0
				let fLeft += fGot
				let iCells := player.AuxVarGetFlt "*_fpCellN"
				player.AuxVarSetFlt "_fpDead" 0
				player.AuxVarSetFlt "_fpWarn" 0
				if eval (player.AuxVarGetFlt "_fpCellOne") == 1
					let iSnd := player.AuxVarGetFlt "_fpSnd"
					if iSnd < 1 || iSnd > 3
						let iSnd := 3
					endif
					let iSnd += 1525397
					let rSnd := BuildRef 0 iSnd
					if eval IsFormValid rSnd
						PlaySound rSnd
					endif
					let fGot := call rCell 0 2
					MessageEx "Cell spent, into your pack as a drained cell. %.0f cells of fuel left." fGot
				else
					MessageEx "No fission batteries: the suit drained %.0f energy weapon cells for power." iCells
				endif
			endif
		endif
		if fLeft < 0
			let fLeft := 0
		endif
		if iUsed > 0
			; the battle rifle's own reload sound (This Machine): a crisp spring snap by default
			; (_fpSnd 3 = WPNThisMachineReloadPt3; 1 and 2 are the other parts of its reload)
			let iSnd := player.AuxVarGetFlt "_fpSnd"
			if iSnd < 1 || iSnd > 3
				let iSnd := 3
			endif
			let iSnd += 1525397	; WPNThisMachineReloadPt1-3 are FalloutNV.esm 0x174696-8 (1525398-400)
			let rSnd := BuildRef 0 iSnd
			if eval IsFormValid rSnd
				PlaySound rSnd
			endif
			player.AuxVarSetFlt "_fpDead" 0
			player.AuxVarSetFlt "_fpWarn" 0
			if iUsed == 1
				MessageEx "Fission battery spent and ejected. %.0f spare(s) left." iSpare
			else
				MessageEx "The trip used %.0f fission batteries. %.0f spare(s) left." iUsed iSpare
			endif
		endif
		if eval fLeft <= 0 && (player.AuxVarGetFlt "_fpDead") == 0
			player.AuxVarSetFlt "_fpDead" 1
			MessageEx "Your power armor is out of power: the servos are dead weight."
		endif
	endif
	player.AuxVarSetFlt "_fpLeft" fLeft
	; the suit's food and water reservoirs (Suit.txt), powered or not: only filtering irradiated food
	; and water takes power (it checks "_fpLeft" itself)
	let rSnd := CompileScript "FissionPower/Suit.txt"
	call rSnd
	; the worn suit carries its processor and reservoirs (SuitWeight.txt)
	let rSnd := CompileScript "FissionPower/SuitWeight.txt"
	call rSnd rSuit 1
	if fLeft > 0
		call rFn 0 rSuit
	else
		call rFn 1 rSuit
	endif

	; low-charge warning, once per battery, when there is no spare to swap in
	let fPct := fLeft / fLife
	if fPct < 0.1 && fLeft > 0 && iSpare == 0
		if eval (player.AuxVarGetFlt "_fpWarn") == 0
			player.AuxVarSetFlt "_fpWarn" 1
			let fGot := 0
			if eval (player.AuxVarGetFlt "_fpAmmo") == 1
				let rCell := CompileScript "FissionPower/Cells.txt"
				let fGot := call rCell 0 0
			endif
			if fGot > 0
				let fGot := fGot / fLife * 100
				MessageEx "Power armor battery below 10 percent and no spare fission batteries: your energy cells hold about %.0f percent of a battery." fGot
			else
				MessageEx "Power armor battery below 10 percent, and no spare fission batteries."
			endif
		endif
	endif
	let fPct *= 1000
	SetUIFloat "HUDMainMenu/_NMPower" fPct
	; the HUD's spare count: batteries, or once they're gone, cells of fuel
	if iSpare == 0
		if eval (player.AuxVarGetFlt "_fpAmmo") == 1
			let rCell := CompileScript "FissionPower/Cells.txt"
			let iSpare := call rCell 0 2
		endif
	endif
	SetUIFloat "HUDMainMenu/_NMBatt" iSpare
end
