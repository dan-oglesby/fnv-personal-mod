; Fission Power: ejects a spent battery (rItem) over your left shoulder, so you see the suit throw it
; out, like a spent shell from a rifle (2026-11-18). It appears at shoulder height a little to your
; left (created there, not moved there: see below); once its model is in, Toss.txt sends it forward, out to the left and up, tumbling, and it
; lands a step or two away. Returns the battery's reference.
; Settings, in metres per second:
;   _fpEjF  forward (default 1.2)
;   _fpEjL  to your left (default 2)
;   _fpEjU  up (default 2.5)
;   _fpEjOff 1  no toss: dropped at your feet, as before
ref rItem
ref rRef
ref rToss
ref rWhere
float fAng
float fEast
float fNorth
float fX
float fY
float fZ

begin Function {rItem}
	if eval (player.AuxVarGetFlt "_fpEjOff") == 1
		let rRef := player.PlaceAtMe rItem 1
		SetFunctionValue rRef
		return
	endif
	; heading: 0 is north (+Y), clockwise. Forward is (sin, cos), your left is (-cos, sin)
	let fAng := player.GetAngle Z
	let fEast := dSin fAng
	let fNorth := dCos fAng
	; at the left shoulder: 22 units (31 cm) to the left, 6 forward, 112 up. Created right there with
	; JIP's PlaceAtCell (2026-11-19): placing it at your feet and then moving it up with MoveTo left
	; its physics frozen, so it hung in midair where it was put and the toss never moved it.
	let fX := player.GetPos X
	let fY := player.GetPos Y
	let fZ := player.GetPos Z
	let fX += -22 * fNorth + 6 * fEast
	let fY += 22 * fEast + 6 * fNorth
	let fZ += 112
	let rWhere := player.GetParentWorldspace
	if eval (IsFormValid rWhere) == 0
		let rWhere := player.GetParentCell
	endif
	let rRef := PlaceAtCell rItem 1 rWhere fX fY fZ
	if eval (IsFormValid rRef) == 0
		let rRef := player.PlaceAtMe rItem 1	; dropped at your feet instead
		SetFunctionValue rRef
		return
	endif
	SetFunctionValue rRef
	let rToss := CompileScript "FissionPower/Toss.txt"
	if eval IsFormValid rToss
		CallAfterSeconds 0.05 rToss 0 rRef 0
	endif
end
