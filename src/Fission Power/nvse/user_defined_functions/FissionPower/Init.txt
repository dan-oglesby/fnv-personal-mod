; Fission Power: default settings, set once per save (JIP auxiliary variables on the player).
; Change them from the console, e.g. player.AuxVarSetFlt "_fpLife" 20000
float fRes

begin Function {}
	if eval (player.AuxVarGetFlt "_fpInit") >= 13
		return
	endif
	; version 13 (2026-11-19): standing in water or swimming fills the suit's water tank (Suit.txt)
	player.AuxVarSetFlt "_fpDraw" 1
	if eval (player.AuxVarGetFlt "_fpInit") == 12
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	; version 12 (2026-11-19): food and water reservoirs (Suit.txt) replace Nutrient Paste. What the
	; paste reservoir held goes into the food reservoir (both count hunger points); the settings of the
	; paste, its mulching, and the bottles the suit left you go.
	let fRes := player.AuxVarGetFlt "_fpPaste"
	let fRes += player.AuxVarGetFlt "_fpFood"
	player.AuxVarSetFlt "_fpFood" fRes
	let fRes := player.AuxVarGetFlt "_fpPasteMax"
	if fRes <= 0
		let fRes := 400
	endif
	player.AuxVarSetFlt "_fpFoodMax" fRes
	player.AuxVarSetFlt "_fpTake" 1
	player.AuxVarErase "_fpPaste"
	player.AuxVarErase "_fpPasteMax"
	player.AuxVarErase "_fpPasteRem"
	player.AuxVarErase "_fpMulch"
	player.AuxVarErase "_fpMulchSeen"
	player.AuxVarErase "_fpPump"
	player.AuxVarErase "_fpWaterKeep"
	player.AuxVarErase "_fpFeedMsg"
	if eval (player.AuxVarGetFlt "_fpInit") >= 10
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	; version 10 (2026-11-18): a spent battery is tossed over your left shoulder (Eject.txt), at these
	; speeds in metres per second: forward, to your left, up. _fpEjOff 1 drops it at your feet instead.
	player.AuxVarSetFlt "_fpEjF" 1.2
	player.AuxVarSetFlt "_fpEjL" 2
	player.AuxVarSetFlt "_fpEjU" 2.5
	if eval (player.AuxVarGetFlt "_fpInit") == 9
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	player.AuxVarSetFlt "_fpCellOne" 1	; version 9: energy cells one at a time, each ejected when spent
	if eval (player.AuxVarGetFlt "_fpInit") == 8
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	; version 8 (2026-11-17): the suit's water reservoir and the weight of its processor and
	; reservoirs (SuitWeight.txt)
	player.AuxVarSetFlt "_fpWaterMax" 400
	player.AuxVarSetFlt "_fpFilter" 2
	player.AuxVarSetFlt "_fpProc" 5
	if eval (player.AuxVarGetFlt "_fpInit") == 7
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	; version 7 (2026-11-17): energy weapon ammo powers the suit after fission batteries (Cells.txt)
	player.AuxVarSetFlt "_fpAmmo" 1
	if eval (player.AuxVarGetFlt "_fpInit") == 6
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	; version 6 (2026-11-17): the suit feeds you (Suit.txt)
	player.AuxVarSetFlt "_fpFeed" 1
	if eval (player.AuxVarGetFlt "_fpInit") == 5
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	player.AuxVarSetFlt "_fpFT" 1	; added in settings version 5: fast travel drains the battery
	if eval (player.AuxVarGetFlt "_fpInit") == 4
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	if eval (player.AuxVarGetFlt "_fpInit") == 3
		player.AuxVarSetFlt "_fpSnd" 3		; added in settings version 4
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	if eval (player.AuxVarGetFlt "_fpInit") == 1
		player.AuxVarSetFlt "_fpSlow" 25	; added in settings version 2
	endif
	if eval (player.AuxVarGetFlt "_fpInit") >= 1
		player.AuxVarSetFlt "_fpIdle" 1		; added in settings version 3
		player.AuxVarSetFlt "_fpSnd" 3		; added in settings version 4
		player.AuxVarSetFlt "_fpInit" 13
		return
	endif
	player.AuxVarSetFlt "_fpInit" 13
	player.AuxVarSetFlt "_fpSnd" 3		; eject sound: battle rifle reload part 3 (the spring snap)
	player.AuxVarSetFlt "_fpLife" 10000	; steps one battery powers a suit for (a step = 52.5 units, 0.75 m)
	player.AuxVarSetFlt "_fpLeft" 10000	; your first suit comes with a charged battery in it
	player.AuxVarSetFlt "_fpPen" 1		; a suit out of power: Agility -2, no Strength bonus, and slower
	player.AuxVarSetFlt "_fpSlow" 25	; how much slower, in percent of movement speed
	player.AuxVarSetFlt "_fpIdle" 1		; standby draw while worn: steps' worth per second (about 2.8 h a battery)
end
