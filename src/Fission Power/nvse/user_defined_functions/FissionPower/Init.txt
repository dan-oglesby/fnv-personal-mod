; Fission Power: default settings, set once per save (JIP auxiliary variables on the player).
; Change them from the console, e.g. player.AuxVarSetFlt "_fpLife" 20000
begin Function {}
	if eval (player.AuxVarGetFlt "_fpInit") >= 7
		return
	endif
	; version 7 (2026-11-17): energy weapon ammo powers the suit after fission batteries (Cells.txt)
	player.AuxVarSetFlt "_fpAmmo" 1
	if eval (player.AuxVarGetFlt "_fpInit") == 6
		player.AuxVarSetFlt "_fpInit" 7
		return
	endif
	; version 6 (2026-11-17): the suit's food system (Suit.txt): a 400-hunger paste reservoir,
	; mulching at 1 step of battery per hunger point, feeding on
	player.AuxVarSetFlt "_fpPasteMax" 400
	player.AuxVarSetFlt "_fpMulch" 1
	player.AuxVarSetFlt "_fpFeed" 1
	if eval (player.AuxVarGetFlt "_fpInit") == 5
		player.AuxVarSetFlt "_fpInit" 7
		return
	endif
	player.AuxVarSetFlt "_fpFT" 1	; added in settings version 5: fast travel drains the battery
	if eval (player.AuxVarGetFlt "_fpInit") == 4
		player.AuxVarSetFlt "_fpInit" 7
		return
	endif
	if eval (player.AuxVarGetFlt "_fpInit") == 3
		player.AuxVarSetFlt "_fpSnd" 3		; added in settings version 4
		player.AuxVarSetFlt "_fpInit" 7
		return
	endif
	if eval (player.AuxVarGetFlt "_fpInit") == 1
		player.AuxVarSetFlt "_fpSlow" 25	; added in settings version 2
	endif
	if eval (player.AuxVarGetFlt "_fpInit") >= 1
		player.AuxVarSetFlt "_fpIdle" 1		; added in settings version 3
		player.AuxVarSetFlt "_fpSnd" 3		; added in settings version 4
		player.AuxVarSetFlt "_fpInit" 7
		return
	endif
	player.AuxVarSetFlt "_fpInit" 7
	player.AuxVarSetFlt "_fpSnd" 3		; eject sound: battle rifle reload part 3 (the spring snap)
	player.AuxVarSetFlt "_fpLife" 10000	; steps one battery powers a suit for (a step = 52.5 units, 0.75 m)
	player.AuxVarSetFlt "_fpLeft" 10000	; your first suit comes with a charged battery in it
	player.AuxVarSetFlt "_fpPen" 1		; a suit out of power: Agility -2, no Strength bonus, and slower
	player.AuxVarSetFlt "_fpSlow" 25	; how much slower, in percent of movement speed
	player.AuxVarSetFlt "_fpIdle" 1		; standby draw while worn: steps' worth per second (about 2.8 h a battery)
end
