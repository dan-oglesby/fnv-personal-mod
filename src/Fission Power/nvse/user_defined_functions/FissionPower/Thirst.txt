; Fission Power: how much water a drink gives the suit's reservoir (2026-11-17): its largest
; RestoreDehydrationLevel magnitude (FalloutNV.esm 0x161F36, decimal 1449782), or 0 if the suit
; won't take it. Taken: water only ("water" in its name: purified, dirty, irradiated and the
; like; not Nuka-Cola or Sunset Sarsaparilla, whose star caps a quest wants), more thirst than
; hunger, not medicine, nothing addictive, no quest items. Irradiated water is taken too and filtered, at more power: then
; "*_fpDirty" is 1.
int iK
int iN
float fHunger
float fThirst
float fMag
ref rItem
ref rEff
ref rHunger
ref rThirst
ref rRad
string_var sName

begin Function {rItem}
	SetFunctionValue 0
	player.AuxVarSetFlt "*_fpDirty" 0
	if eval (GetType rItem) != 47
		return
	endif
	if eval (GetIngestibleFlag rItem 4) || (GetAddictionChance rItem) > 0 || (IsQuestItem rItem)
		return
	endif
	let sName := GetName rItem
	if eval (sv_Find "water" sName) < 0
		return
	endif
	let rHunger := BuildRef 0 1453004
	let rThirst := BuildRef 0 1449782
	let rRad := BuildRef 0 132374
	let fHunger := 0
	let fThirst := 0
	let iN := GetNumEffects rItem
	let iK := 0
	while iK < iN
		let rEff := GetNthEffectBase rItem iK
		let fMag := GetNthEffectTraitNumeric rItem iK 0
		if eval rEff == rRad
			player.AuxVarSetFlt "*_fpDirty" 1
		elseif eval rEff == rHunger
			if fMag > fHunger
				let fHunger := fMag
			endif
		elseif eval rEff == rThirst
			if fMag > fThirst
				let fThirst := fMag
			endif
		endif
		let iK += 1
	loop
	if fThirst <= fHunger
		return
	endif
	SetFunctionValue fThirst
end
