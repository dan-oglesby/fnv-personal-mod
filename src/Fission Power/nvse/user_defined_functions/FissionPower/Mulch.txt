; Fission Power: which carried food the suit mulches next (2026-11-18, moved out of Suit.txt).
; Of the food you carry whose nutrition (Nutri.txt) fits in fRoom, the one with the most weight per
; hunger point (so the most weight leaves your pack); never a quest item. Returns it, or nothing,
; and leaves its nutrition in "*_fpMulchMag".
; Why a function of its own: in Suit.txt this loop hit a bad statement read on every run (the
; engine's "Unable to find function definition for command -27904", 5,000+ times a session), and
; the failure ended Suit.txt before it saved its timer and reservoir, so it ran again (and failed
; again) every second. Here the loop has no continue statements, and Suit.txt saves its state
; before calling it, so a failure costs at most one try per 10 seconds.
float fRoom
float fMag
float fScore
float fBest
float fBestMag
float fWt
int iNum
int iIdx
ref rItem
ref rBest
ref rFn

begin Function {fRoom}
	let rFn := CompileScript "FissionPower/Nutri.txt"
	let fBest := -1
	let fBestMag := 0
	let iNum := player.GetNumItems
	let iIdx := 0
	while iIdx < iNum
		let rItem := player.GetInventoryObject iIdx
		let iIdx += 1
		if eval IsFormValid rItem
			if eval (GetType rItem) == 47
				if eval (IsQuestItem rItem) == 0
					let fMag := call rFn rItem
					if eval fMag > 0 && fMag <= fRoom
						let fWt := GetWeight rItem
						let fScore := fWt / fMag
						if eval fScore > fBest
							let fBest := fScore
							let fBestMag := fMag
							let rBest := rItem
						endif
					endif
				endif
			endif
		endif
	loop
	player.AuxVarSetFlt "*_fpMulchMag" fBestMag
	if eval IsFormValid rBest
		SetFunctionValue rBest
	else
		SetFunctionValue 0	; always a value: the caller stores it in a ref
	endif
end
