; Wasteland Market: why a category's price last moved.
int iWhy
string_var sName

begin Function {iWhy}
	let sName := "settling"
	if iWhy == 1
		let sName := "your buying"
	elseif iWhy == 2
		let sName := "your selling"
	elseif iWhy == 3
		let sName := "a scavenger's big haul flooded the market"
	elseif iWhy == 4
		let sName := "a shortage"
	elseif iWhy == 5
		let sName := "traders are bringing in stock"
	elseif iWhy == 6
		let sName := "a caravan came in"
	elseif iWhy == 7
		let sName := "a caravan was lost on the road"
	endif
	SetFunctionValue sName
end
