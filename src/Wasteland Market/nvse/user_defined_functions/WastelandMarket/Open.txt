; Wasteland Market: the Mojave Market Pamphlet.
; - With Wasteland Bank (2026-11-17): it's the Mojave Market Report, a weekly paper the courier squads
;   hand you when you meet them (WastelandBank/Report.txt), dated, with the war's news and the
;   markets as they stood that day. The last four editions are kept ("_wmEd" 0-3, newest first;
;   "_wmEdN"); "*_wmEdView" is the one shown. Before the first, it says how to get one.
; - Without it: the market as it is now (Body.txt).
; Buttons are listed as they apply and noted in "*_wmBtn" (0 older edition, 1 newer, 2 market events
; on or off, 3 close) for OpenBtn.txt.
int iN
int iV
int iB
int iOlder
string_var sBody
string_var sTitle
string_var sEvOn
string_var sBtns
ref rFn
ref rCb

begin Function {}
	let sEvOn := "Market events: turn off"
	if eval (player.AuxVarGetFlt "_wmEvents") == 0
		let sEvOn := "Market events: turn on"
	endif
	player.AuxVarErase "*_wmBtn"
	let sBtns := ""
	let iB := 0
	if eval IsModLoaded "WastelandBank.esp"
		let sTitle := "Mojave Market Report"
		let iN := player.AuxVarGetFlt "_wmEdN"
		if iN == 0
			let sBody := "No edition yet. The bank's courier squads hand you each week's report when you meet them."
		else
			let iV := player.AuxVarGetFlt "*_wmEdView"
			if iV >= iN
				let iV := iN - 1
			endif
			let sBody := player.AuxVarGetStr "_wmEd" iV
			if iV > 0
				let sBody := (sv_Construct "(An older edition: %.0f back.)%r" iV) + sBody
			endif
			let iOlder := iV + 1
			if iOlder < iN
				let sBtns := sBtns + "|Older edition"
				player.AuxVarSetFlt "*_wmBtn" 0 iB
				let iB += 1
			endif
			if iV > 0
				let sBtns := sBtns + "|Newer edition"
				player.AuxVarSetFlt "*_wmBtn" 1 iB
				let iB += 1
			endif
		endif
	else
		let sTitle := "Mojave Market Pamphlet"
		let rFn := CompileScript "WastelandMarket/Body.txt"
		let sBody := call rFn
		let sBody := (sv_Construct "Prices across the Mojave, as merchants quote them this week.%r%r") + sBody
	endif
	let sBtns := sBtns + "|" + sEvOn + "|Close"
	player.AuxVarSetFlt "*_wmBtn" 2 iB
	let iB += 1
	player.AuxVarSetFlt "*_wmBtn" 3 iB
	let rCb := CompileScript "WastelandMarket/OpenBtn.txt"
	MessageBoxExAlt rCb "^%z^%z%z" sTitle sBody sBtns
end
