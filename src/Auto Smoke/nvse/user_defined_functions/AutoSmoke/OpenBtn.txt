; Auto Smoke: menu buttons (0 toggle auto-smoke; 1 cycle when to light up:
; keep lit > 1 > 3 > 5 > 10 > 20 min > only when craving > keep lit; 2 toggle opening packs;
; 3 smoke one now; 4 close).
int iBtn
int iGap
int iRes
ref rNext

begin Function {iBtn}
	if iBtn == 4
		return
	endif
	if iBtn == 0
		if eval (player.AuxVarGetFlt "_asOn") == 1
			player.AuxVarSetFlt "_asOn" 0
			print "Auto Smoke: off"
		else
			player.AuxVarSetFlt "_asOn" 1
			print "Auto Smoke: on"
		endif
	elseif iBtn == 1
		let iGap := player.AuxVarGetFlt "_asGap"
		if iGap < 0
			let iGap := 0
		elseif iGap == 0
			let iGap := 1
		elseif iGap == 1
			let iGap := 3
		elseif iGap == 3
			let iGap := 5
		elseif iGap == 5
			let iGap := 10
		elseif iGap == 10
			let iGap := 20
		else
			let iGap := -1
		endif
		player.AuxVarSetFlt "_asGap" iGap
	elseif iBtn == 2
		if eval (player.AuxVarGetFlt "_asPacks") == 1
			player.AuxVarSetFlt "_asPacks" 0
		else
			player.AuxVarSetFlt "_asPacks" 1
		endif
	elseif iBtn == 3
		let rNext := CompileScript "AutoSmoke/IsLit.txt"
		if eval (call rNext) == 1
			MessageEx "You're already smoking."
			return
		endif
		let rNext := CompileScript "AutoSmoke/LightUp.txt"
		let iRes := call rNext
		if iRes == 1
			MessageEx "Opened a pack. Choose Smoke one now again to light up."
		elseif iRes == 2
			MessageEx "Take off whatever is covering your mouth first."
		elseif iRes == 3
			MessageEx "Nothing to smoke."
		endif
		return
	endif
	let rNext := CompileScript "AutoSmoke/Open.txt"
	call rNext
end
