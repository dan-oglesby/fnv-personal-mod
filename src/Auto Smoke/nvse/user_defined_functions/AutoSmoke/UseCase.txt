; Auto Smoke: the case was equipped. Takes it off again and opens the settings. Checking that it
; is still worn means a double equip opens the menu only once.
int iPlugin
ref rCase
ref rOpen

begin Function {}
	let iPlugin := GetModIndex "AutoSmoke.esp"
	let rCase := BuildRef iPlugin 2048	; ASCigaretteCase
	if eval (player.GetEquipped rCase) == 0
		return
	endif
	player.UnequipItem rCase 0 1
	if player.GetDead
		return
	endif
	let rOpen := CompileScript "AutoSmoke/Open.txt"
	call rOpen
end
