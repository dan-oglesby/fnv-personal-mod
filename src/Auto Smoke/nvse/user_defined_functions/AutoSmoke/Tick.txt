; Auto Smoke: every 5 s of play (registered by ln_auto_smoke.txt; not in menus or paused).
; 1. Nicotine alertness (Hardcore, if on and its sleep script compiled): a smoke that has just been
;    lit, by hand or automatically, gets NicSmoke; NicTick handles withdrawal and bedtime.
; 2. Auto-smoke (if on): when you are not smoking, counts the time since the last smoke went out
;    and lights the next once the chosen break has passed; or in "only when craving" mode as soon as
;    you have nicotine withdrawal; or in "when tired" mode once you are Tired (sleep deprivation
;    200+) and tolerance would still give at least half the effect (at most 1 smoke in 6 hours).
int iIdle
int iGap
int iRes
int iKind
int iWas
int iN
float fSlp
ref rFn
ref rEff

begin Function {}
	if player.GetDead
		return
	endif
	let rFn := CompileScript "AutoSmoke/LitKind.txt"
	let iKind := call rFn
	let iWas := player.AuxVarGetFlt "_asWasLit"
	player.AuxVarSetFlt "_asWasLit" iKind

	if eval (player.AuxVarGetFlt "_asNicOk") == 1
		if (iKind > 0) && (iWas == 0)
			let rFn := CompileScript "AutoSmoke/NicSmoke.txt"
			call rFn iKind
		endif
		let rFn := CompileScript "AutoSmoke/NicTick.txt"
		call rFn
	endif

	if eval (player.AuxVarGetFlt "_asOn") == 0
		return
	endif
	if iKind > 0
		player.AuxVarSetFlt "_asIdle" 0
		return
	endif
	let iIdle := player.AuxVarGetFlt "_asIdle"
	let iIdle += 5
	player.AuxVarSetFlt "_asIdle" iIdle
	let iGap := player.AuxVarGetFlt "_asGap"
	if iGap == -1
		let rEff := BuildRef 0 1026831	; AddictRedAGTobacco 0x0FAB0F, part of Nicotine Withdrawal
		if eval (player.HasMagicEffect rEff) == 0
			return
		endif
	elseif iGap == -2
		if eval (player.AuxVarGetFlt "_asNicOk") == 0
			return
		endif
		if eval (player.IsHardcore) == 0
			return
		endif
		let rFn := CompileScript "AutoSmoke/NicSleep.txt"
		let fSlp := call rFn 0 0
		if fSlp < 200
			return
		endif
		let rFn := CompileScript "AutoSmoke/NicCount.txt"
		let iN := call rFn
		if iN > 1
			return
		endif
	elseif iIdle < iGap * 60
		return
	endif
	let rFn := CompileScript "AutoSmoke/LightUp.txt"
	let iRes := call rFn
	if iRes == 0
		player.AuxVarSetFlt "_asIdle" 0
	endif
end
