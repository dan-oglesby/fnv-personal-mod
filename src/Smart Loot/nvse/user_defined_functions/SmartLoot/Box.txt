; Smart Loot: loots one container or corpse (rRef), moving at most iBudget stacks. Returns how many
; stacks it moved. Containers that are owned, or in an owned cell, are never touched, unless every
; owner is your enemy (Owned.txt). Moving items by script is no crime and costs no karma.
; A locked, unowned container is unlocked first if the setting allows: with its key, or with your
; Lockpick skill if the lock is no harder than your skill (and, if set, a bobby pin), for the same
; XP as Auto Loot Ultimate gives (20, plus 10 per lock level). "Requires key" locks need the key.
; Items keep their condition and mods: RemoveItemTarget (JIP) moves them whole.
int iType
int iLock
int iSet
int iPins
int iNum
int iIdx
int iHave
int iTake
int iMoved
int iXP
float fSkill
ref rFn
ref rItem
ref rPin
ref rSnd
ref rRef
int iBudget
int iNow
int iPrev
int iVisit
int iLog
int iDead
int iLockd
string_var sLog

begin Function {rRef, iBudget}
	SetFunctionValue 0
	if eval (IsFormValid rRef) == 0
		return
	endif
	let iType := GetType (rRef.GetBaseObject)
	; debug trace (_slDebug 1): each corpse or container once, with every item decision
	let iLog := 0
	if eval (player.AuxVarGetFlt "_slDebug") == 1 && (rRef.AuxVarGetFlt "*_slSeen") == 0
		rRef.AuxVarSetFlt "*_slSeen" 1
		let iLog := 1
		let iDead := rRef.GetDead
		let iLockd := rRef.GetLocked
		let iNum := rRef.GetNumItems
		let sLog := sv_Construct "%i %n: type %.0f, dead %.0f, locked %.0f, %.0f item types" rRef rRef iType iDead iLockd iNum
		PrintF "SmartLoot.log" sLog
	endif
	if iType == 27
		let rFn := CompileScript "SmartLoot/Owned.txt"
		if eval (call rFn rRef) == 1
			if iLog == 1
				PrintF "SmartLoot.log" "  owned (or in an owned place): left alone"
			endif
			return
		endif
	endif
	if eval rRef.GetLocked
		let iSet := player.AuxVarGetFlt "_slLock"
		if iSet == 0
			return
		endif
		let iLock := rRef.GetLockLevel
		if eval rRef.PlayerHasKey
			rRef.Unlock
		elseif iLock <= 100
			let fSkill := player.GetAV Lockpick
			if fSkill < iLock
				return
			endif
			if iSet == 2
				let rPin := BuildRef 0 10	; Bobby Pin (FalloutNV.esm 0x0000000A, editor ID Lockpick)
				let iPins := player.GetItemCount rPin
				if iPins < 1
					return
				endif
				player.RemoveItem rPin 1 1
			endif
			rRef.Unlock
			let iXP := 20 + (Floor (iLock / 25)) * 10
			player.RewardXP iXP
			if eval (player.AuxVarGetFlt "_slMsg") >= 1
				MessageEx "Smart Loot: unlocked %n (lock %.0f)" rRef iLock
			endif
		else
			return
		endif
	endif

	let rFn := CompileScript "SmartLoot/Judge.txt"
	; a "visit": the container was out of reach on the previous check and is in reach now
	if iType == 27
		let iNow := player.AuxVarGetFlt "*_slTickN"
		let iPrev := rRef.AuxVarGetFlt "*_slSeenT"
		rRef.AuxVarSetFlt "*_slSeenT" iNow
		if iNow - iPrev > 1
			rRef.AuxVarSetFlt "*_slVisit" iNow
		endif
	endif
	let iMoved := 0
	let iNum := rRef.GetNumItems
	let iIdx := iNum - 1
	while iIdx >= 0
		if iMoved >= iBudget
			break
		endif
		let rItem := rRef.GetInventoryObject iIdx
		let iIdx -= 1
		if eval (IsFormValid rItem) == 0
			continue
		endif
		let iHave := rRef.GetItemCount rItem
		let iTake := call rFn rItem iHave
		if iLog == 1
			let sLog := sv_Construct "  %i %n x%.0f: take %.0f" rItem rItem iHave iTake
			PrintF "SmartLoot.log" sLog
		endif
		if iTake > 0
			rRef.RemoveItemTarget rItem player iTake 1	; 1 = clear the old owner, so nothing arrives marked stolen
			let rSnd := CompileScript "SmartLoot/Sound.txt"
			call rSnd rItem
			let rSnd := CompileScript "SmartLoot/Note.txt"
			call rSnd rItem
			let iMoved += 1
		endif
	loop
	; searched all the way through (not stopped by the budget): show it, once per visit
	if iType == 27 && iIdx < 0
		let iVisit := rRef.AuxVarGetFlt "*_slVisit"
		if eval (rRef.AuxVarGetFlt "*_slAjarV") != iVisit
			rRef.AuxVarSetFlt "*_slAjarV" iVisit
			let rSnd := CompileScript "SmartLoot/Ajar.txt"
			call rSnd rRef
		endif
	endif
	SetFunctionValue iMoved
end
