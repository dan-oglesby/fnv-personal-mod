; Smart Loot: settings, page 3 (survival, what it uses for you). No colon in the menu title: the
; first version had one and failed to precompile (2026-11-15), the only difference from the pages
; that compiled.
float fVal
string_var sEat
string_var sNap
string_var sRadAway
string_var sStim
string_var sBag
string_var sChoice
string_var sBody
ref rCb

begin Function {}
	let sEat := "off"
	if eval (player.AuxVarGetFlt "_seFood") == 1
		let sEat := "on"
	endif
	let sNap := "off"
	if eval (player.AuxVarGetFlt "_seSleep") == 1
		let sNap := "on"
	endif
	let sRadAway := "off"
	if eval (player.AuxVarGetFlt "_seRads") == 1
		let sRadAway := "on"
	endif
	let fVal := player.AuxVarGetFlt "_seHP"
	let sStim := "never"
	if fVal > 0
		let sStim := sv_Construct "at %.0f percent health or less, even in combat" fVal
	endif
	let sBag := "off"
	if eval (player.AuxVarGetFlt "_seLimb") == 1
		let sBag := "on"
	endif
	let fVal := player.AuxVarGetFlt "_seRad"
	let sChoice := "irradiated food only when nothing clean is left, never anything addictive"
	if fVal == 0
		let sChoice := "never anything irradiated or addictive"
	elseif fVal == 2
		let sChoice := "anything, irradiated or addictive included"
	endif

	let fVal := player.AuxVarGetFlt "_seEarly"
	let sBody := sv_Construct "Out of combat, steps in %.0f points before the first penalty of each need.%r" fVal
	let sBody := sBody + (sv_Construct "Water and food (Hardcore): %z. Sleep items such as coffee: %z.%r" sEat sNap)
	let sBody := sBody + (sv_Construct "RadAway near radiation sickness, Rad-X while taking radiation: %z.%r" sRadAway)
	let sBody := sBody + (sv_Construct "Stimpaks: %z. Doctor's bag for a crippled limb: %z.%r" sStim sBag)
	let sBody := sBody + (sv_Construct "Choices: %z. Food and drink: the heaviest for what it gives first. Aid: the cheapest that works." sChoice)
	if eval (player.AuxVarGetFlt "_seOk") == 0
		let sBody := sBody + (sv_Construct "%r")
		let sBody := sBody + "Unavailable, because its script did not load. It needs ITR NVSE, see the console."
	endif
	let rCb := CompileScript "SmartLoot/Open3Btn.txt"
	MessageBoxExAlt rCb "^Smart Loot survival^%z|Water and food on or off|Change how early|Sleep items on or off|RadAway and Rad-X on or off|Change stimpak health|Doctor's bag on or off|Change choices|Back to the start|Close" sBody
end
