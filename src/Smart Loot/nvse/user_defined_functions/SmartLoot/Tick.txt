; Smart Loot: the regular check (every 1.5 s of play; registered by ln_smart_loot.txt, never in
; menus or while paused). Skipped while dead or, unless set otherwise, in combat.
; 1. Rebuilds the want-lists (Want.txt) when the number of stacks you carry changes, or every 8th
;    check.
; 2. One scan of references within reach (xNVSE GetRefs with type 0, which skips the typed matcher
;    that crashes on a reference with no base item), sorted into loose items and containers or
;    corpses. Nothing is picked up during the scan: picking up mid-scan is what crashed Auto Loot
;    Enhanced.
; 3. Then up to 10 pickups (an enemy's things are made yours first, so it's no theft): loose items by activating them for you (the same as pressing E: their
;    scripts run and the game adds them as usual), then containers and corpses (Box.txt).
int iInv
int iT
int iActs
int iType
int iHave
int iTake
int iMsg
int iLeft
int iLvl
int iTickN
int iDead
string_var sLog
int iOwn
float fRad
float fX
array_var aRefs
array_var aEntry
array_var aLoose
array_var aBoxes
array_var aPers
ref rRef
ref rBase
ref rFn
ref rJudge
ref rOwned
ref rSnd
ref rNote
ref rWS
ref rPC
ref rAlways

begin Function {}
	if player.GetDead
		return
	endif
	; nothing during VATS playback (JIP IsInKillCam): using or equipping items mid-VATS can leave the
	; VATS camera stuck (2026-11-15, a Mysterious Stranger appearance froze while a radscorpion closed in)
	if IsInKillCam
		return
	endif
	; auto-consume first: stimpaks must work in combat (Survive.txt waits out combat for the rest)
	if eval (player.AuxVarGetFlt "_seOk") == 1
		let rFn := CompileScript "SmartLoot/Survive.txt"
		call rFn
	endif
	if eval (player.AuxVarGetFlt "_slOn") == 0
		return
	endif
	if eval (player.AuxVarGetFlt "_slFight") == 0
		if player.IsInCombat
			if eval (player.AuxVarGetFlt "_slDebug") == 1 && (player.AuxVarGetFlt "*_slFightLog") == 0
				player.AuxVarSetFlt "*_slFightLog" 1
				PrintF "SmartLoot.log" "paused: you are in combat"
			endif
			return
		endif
	endif
	if eval (player.AuxVarGetFlt "*_slFightLog") == 1
		player.AuxVarSetFlt "*_slFightLog" 0
		if eval (player.AuxVarGetFlt "_slDebug") == 1
			PrintF "SmartLoot.log" "resumed: out of combat"
		endif
	endif

	; one pass at a time (2026-11-18): when FastTick.txt ran a pass less than half a second ago (two
	; of its ticks), this timed pass would find nothing new, so it skips the scan
	let fX := player.AuxVarGetFlt "*_slLastF"
	if eval fX > 0 && ((player.AuxVarGetFlt "*_slFastN") - fX) < 2
		return
	endif
	let iInv := player.GetNumItems
	let iT := player.AuxVarGetFlt "*_slT"
	let iT += 1
	if eval (iInv != (player.AuxVarGetFlt "*_slInv")) || iT >= 8
		let rFn := CompileScript "SmartLoot/Want.txt"
		call rFn
		player.AuxVarSetFlt "*_slInv" iInv
		let iT := 0
	endif
	player.AuxVarSetFlt "*_slT" iT
	player.AuxVarSetFlt "*_slSndN" 0
	let iTickN := player.AuxVarGetFlt "*_slTickN"
	let iTickN += 1
	player.AuxVarSetFlt "*_slTickN" iTickN
	; where and when this pass ran, for FastTick.txt's early passes
	let fX := player.GetPos X
	player.AuxVarSetFlt "*_slLX" fX
	let fX := player.GetPos Y
	player.AuxVarSetFlt "*_slLY" fX
	let fX := player.AuxVarGetFlt "*_slFastN"
	player.AuxVarSetFlt "*_slLastF" fX

	let fRad := player.AuxVarGetFlt "_slRad"
	let aRefs := player.GetRefs 0 1 0 fRad
	if eval aRefs == Ar_Null
		let aRefs := Ar_Construct "array"
	endif
	; Persistent actors (quest-placed, like the Legion assassin squads) live outside the cells GetRefs
	; walks. Every loaded actor is in the game's process lists, which only the main thread changes,
	; the thread this runs on; dead ones in reach are added from there (JIP
	; GetActorsByProcessingLevel 0 to 3). Not the worldspace persistent cell: walking that every check
	; raced the loader thread and brought back the 0xA61A74 crash (2026-11-15). Actors found both
	; ways are kept once (Box would otherwise run twice on them).
	let iLvl := 0
	while iLvl < 4
		let aPers := GetActorsByProcessingLevel iLvl
		let iLvl += 1
		if eval aPers == Ar_Null
			continue
		endif
		foreach aEntry <- aPers
			let rRef := aEntry["value"]
			if eval (IsFormValid rRef) == 0 || rRef == player
				continue
			endif
			if eval rRef.GetDead && (player.GetDistance rRef) <= fRad
				Ar_Append aRefs rRef
			endif
		loop
	loop
	; and the dead body under your crosshair, if it's in reach (xNVSE GetCrosshairRef): a corpse
	; neither list returned is still looted when you look at it. A persistent Viper Leader
	; (FalloutNV.esm VViperGangLeader...Pers) wasn't picked up (2026-11-16).
	let rRef := GetCrosshairRef
	if eval IsFormValid rRef
		let iType := GetType (rRef.GetBaseObject)
		if iType >= 42 && iType <= 45
			if eval rRef.GetDead && (player.GetDistance rRef) <= fRad
				Ar_Append aRefs rRef
			endif
		endif
	endif
	let rAlways := BuildRef (GetModIndex "SmartLoot.esp") 2052	; SLAlwaysTake
	let aLoose := Ar_Construct "array"
	let aBoxes := Ar_Construct "array"
	foreach aEntry <- aRefs
		let rRef := aEntry["value"]
		if eval (IsFormValid rRef) == 0 || rRef == player
			continue
		endif
		let rBase := rRef.GetBaseObject
		if eval (IsFormValid rBase) == 0 || rRef.GetDisabled
			continue
		endif
		if eval (player.GetDistance rRef) > fRad
			continue
		endif
		let iType := GetType rBase
		if iType == 27
			Ar_Append aBoxes rRef
		elseif iType >= 42 && iType <= 45
			; actors: NPC 42, creature 43, and 44/45 in case a leveled actor reports its leveled list
			if eval (rRef.AuxVarGetFlt "*_slInT") == iTickN
				continue	; already listed this check (found by both scans)
			endif
			rRef.AuxVarSetFlt "*_slInT" iTickN
			if eval (player.AuxVarGetFlt "_slDebug") == 1 && (rRef.AuxVarGetFlt "*_slSeenA") == 0
				rRef.AuxVarSetFlt "*_slSeenA" 1
				let iDead := rRef.GetDead
				let sLog := sv_Construct "actor %i %n in reach: type %.0f, dead %.0f" rRef rRef iType iDead
				PrintF "SmartLoot.log" sLog
			endif
			if eval rRef.GetDead && (rRef.GetPlayerTeammate == 0)
				Ar_Append aBoxes rRef
			endif
		elseif iType == 24 || iType == 25 || iType == 31 || iType == 40 || iType == 41
			Ar_Append aLoose rRef
		elseif iType == 46 || iType == 47 || iType == 49 || iType == 103 || iType == 116
			Ar_Append aLoose rRef
		elseif eval (ListGetFormIndex rAlways rBase) >= 0
			Ar_Append aLoose rRef	; casino chips, caravan money and the like
		endif
	loop

	let rJudge := CompileScript "SmartLoot/Judge.txt"
	let rOwned := CompileScript "SmartLoot/Owned.txt"
	let rSnd := CompileScript "SmartLoot/Sound.txt"
	let rNote := CompileScript "SmartLoot/Note.txt"
	let iActs := 0
	foreach aEntry <- aLoose
		if iActs >= 10
			break
		endif
		let rRef := aEntry["value"]
		if eval (IsFormValid rRef) == 0 || rRef.GetDisabled
			continue
		endif
		; not while it is still loading (review 2026-11-18): picking up a reference whose model isn't
		; in yet is the half-loaded state a dropped-weapon crash came from (2026-11-16); it is taken
		; on a later check instead
		if eval (rRef.HasLoaded3D) == 0
			continue
		endif
		let rBase := rRef.GetBaseObject
		if eval (IsFormValid rBase) == 0
			continue
		endif
		let iOwn := call rOwned rRef
		if iOwn == 1
			continue
		endif
		let iHave := rRef.GetRefCount
		if iHave < 1
			let iHave := 1
		endif
		let iTake := call rJudge rBase iHave
		; a loose stack is picked up whole: only if all of it is wanted (ammo: any room under the cap)
		if eval iTake >= iHave || (iTake > 0 && (GetType rBase) == 41)
			if iOwn == 2
				rRef.SetOwnership	; an enemy's: make it yours first, so taking it is no theft
			endif
			rRef.Activate player 1
			call rSnd rBase
			call rNote rBase
			let iActs += 1
		endif
	loop

	let rFn := CompileScript "SmartLoot/Box.txt"
	foreach aEntry <- aBoxes
		if iActs >= 10
			break
		endif
		let rRef := aEntry["value"]
		let iLeft := 10 - iActs
		let iTake := call rFn rRef iLeft
		let iActs += iTake
	loop

	let iMsg := player.AuxVarGetFlt "_slMsg"
	if iMsg >= 2 && iActs > 0
		MessageEx "Smart Loot: picked up %.0f item(s)" iActs
	endif
	let aRefs := Ar_Null
	let aPers := Ar_Null
	let aLoose := Ar_Null
	let aBoxes := Ar_Null
	let aEntry := Ar_Null
end
