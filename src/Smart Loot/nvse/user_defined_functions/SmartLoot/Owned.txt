; Smart Loot: may this reference be looted (or unlocked)? Checks its owner and its cell's owner.
;   0  nobody owns it or its cell: take it.
;   2  every owner it has is an enemy of yours (the setting "take from enemies" is on): take it, but
;      first make it yours (Tick.txt / Box.txt) so it isn't theft. No karma is lost: there's no
;      moral code against robbing the Legion when it already wants you dead.
;   1  anything else: leave it. That includes anything you own, so your own containers and the
;      things you put down at home are never looted.
; "Enemy" is the game's own test, GetPCEnemyOfFaction (what the Lucky 38 scripts use). An owner that
; is a person counts as an enemy if any faction they belong to is your enemy.
; Corpses are not checked: looting the dead is never a crime.
ref rRef
ref rOwner
ref rFac
int iStep
int iHostile
int iNum
int iK

begin Function {rRef}
	SetFunctionValue 1
	if eval (IsFormValid rRef) == 0
		return
	endif
	let iHostile := 0
	let iStep := 0
	while iStep < 2
		if iStep == 0
			let rOwner := rRef.GetOwner
		else
			let rOwner := rRef.GetParentCellOwner
		endif
		let iStep += 1
		if eval (IsFormValid rOwner) == 0
			continue
		endif
		; someone owns this: it can only be taken if they're your enemy
		if eval (player.AuxVarGetFlt "_slEnemy") == 0
			return
		endif
		if eval (GetType rOwner) == 8
			if eval (GetPCEnemyOfFaction rOwner) == 0
				return
			endif
		elseif eval (GetType rOwner) == 42
			let iNum := GetBaseNumFactions rOwner
			let iK := 0
			let iHostile := 0
			while iK < iNum
				let rFac := GetBaseNthFaction iK rOwner
				if eval IsFormValid rFac
					if eval GetPCEnemyOfFaction rFac
						let iHostile := 1
						break
					endif
				endif
				let iK += 1
			loop
			if iHostile == 0
				return
			endif
		else
			return
		endif
		let iHostile := 1
	loop
	if iHostile == 1
		SetFunctionValue 2
	else
		SetFunctionValue 0
	endif
end
