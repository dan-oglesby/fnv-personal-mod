; Smart Loot: fills SLAlwaysTake (SmartLoot.esp 0x804), the items always taken whatever the other
; rules say: every currency, NCR dogtags (which the game flags as quest items), skill magazines and
; bobby pins. Run on every load by ln_smart_loot.txt. FalloutNV.esm form IDs, in decimal:
;   caps 0x00000F; NCR $5, $20, $100 0x13A4D6-8; Legion Denarius 0x137B9D, Aureus 0x13A4D9,
;   Denarius (mangled) 0x173236; pre-war money 0x034067; the caravan-money versions of the notes and
;   coins 0x176AC4-9; casino chips: Gomorrah 0x139FE0, The Tops 0x144C3B, Ultra-Luxe 0x144C3C,
;   Vikki and Vance 0x144C3D, Atomic Wrangler 0x144C3E; NCR Dogtag 0x123054, Irradiated NCR Dogtag
;   0x13FE77; skill magazines (2026-11-16): Salesman Weekly 0x0EC4A4, Future Weapons Today to
;   Programmer's Digest 0x0EC4A6-0x0EC4B1, True Police Stories 0x135689. Plus Dead Money's Sierra
;   Madre Chip (DeadMoney.esm 0x011236) if Dead Money is loaded. Bobby Pin 0x00000A (editor ID Lockpick;
;   2026-11-18): weightless and worth nothing, so the value rule never took them.
int iPlug
int iIdx
int iId
ref rList
ref rItem
array_var aIds

begin Function {}
	let iPlug := GetModIndex "SmartLoot.esp"
	let rList := BuildRef iPlug 2052
	ListClear rList
	let aIds := Ar_List 15, 10, 1287382, 1287383, 1287384, 1276829, 1287385, 1520182, 213095, 1534660, 1534661, 1534662, 1534663, 1534664, 1534665
	let iIdx := 0
	while iIdx < Ar_Size aIds
		let iId := aIds[iIdx]
		let rItem := BuildRef 0 iId
		if eval IsFormValid rItem
			ListAddForm rList rItem
		endif
		let iIdx += 1
	loop
	let aIds := Ar_List 1286112, 1330235, 1330236, 1330237, 1330238, 1192020, 1310327
	let iIdx := 0
	while iIdx < Ar_Size aIds
		let iId := aIds[iIdx]
		let rItem := BuildRef 0 iId
		if eval IsFormValid rItem
			ListAddForm rList rItem
		endif
		let iIdx += 1
	loop
	; skill magazines
	let aIds := Ar_List 967844, 967846, 967847, 967848, 967849, 967850, 967851, 967852, 967853, 967854, 967855, 967856, 967857, 1267337
	let iIdx := 0
	while iIdx < Ar_Size aIds
		let iId := aIds[iIdx]
		let rItem := BuildRef 0 iId
		if eval IsFormValid rItem
			ListAddForm rList rItem
		endif
		let iIdx += 1
	loop
	; Fission Power's fuel besides batteries (2026-11-17): energy weapon ammo, marked "*_slFuel" for
	; Judge.txt (taken up to "_slCells" while you own power armor). FalloutNV.esm IDs: Energy Cell
	; 132978 and its Bulk, Over Charge, Max Charge kinds 1409811, 1409759, 1409760; Microfusion Cell
	; 17541, 1409810, 1409803, 1409804; Electron Charge Pack 439614, 1409809, 1409805, 1409806
	if eval IsModLoaded "FissionPower.esp"
		let aIds := Ar_List 132978, 1409811, 1409759, 1409760, 17541, 1409810, 1409803, 1409804, 439614, 1409809, 1409805, 1409806
		let iIdx := 0
		while iIdx < Ar_Size aIds
			let iId := aIds[iIdx]
			let rItem := BuildRef 0 iId
			if eval IsFormValid rItem
				AuxVarSetFlt "*_slFuel" 1 0 rItem
			endif
			let iIdx += 1
		loop
	endif
	; Wasteland Bank's own notes (WastelandBank.esp 0x809, 0x80A)
	if eval IsModLoaded "WastelandBank.esp"
		let iPlug := GetModIndex "WastelandBank.esp"
		let rItem := BuildRef iPlug 2057
		ListAddForm rList rItem
		let rItem := BuildRef iPlug 2058
		ListAddForm rList rItem
	endif
	if eval IsModLoaded "DeadMoney.esm"
		let iPlug := GetModIndex "DeadMoney.esm"
		let rItem := BuildRef iPlug 70198	; NVDLC01SierraMadreChip
		if eval IsFormValid rItem
			ListAddForm rList rItem
		endif
	endif
	let aIds := Ar_Null
end
