; Monocyte Limb Regen (2026-11-11): the Monocyte Breeder implant from the New Vegas Medical
; Clinic should "eventually heal limb injuries at the same pace as the regular healing".
;
; Vanilla: ImplantRegenScript restores Health by ScriptEffectElapsedSeconds / 10 every frame, so
; 0.1 a second (1 HP every 10 s). This restores each limb's condition at the same 0.1 a second
; while you have the implant (ImplantRegenPerk, FalloutNV.esm 0x14CCE1). Limbs run 0-100, so a
; fully crippled limb is whole again in about 17 minutes of play.
;
; Run by JIP LN NVSE's script runner (ln_: new game and every load). xNVSE clears all repeating
; calls on every load and new game, so registering here never stacks. Ticks once a second, only
; in game mode and not while paused (flags 4 + 16), like the implant's own effect script.
; Needs xNVSE 6.2.8+ (CallWhilePerSeconds) and JIP LN 56.95+. The tick is MonocyteRegen/Tick.txt,
; started at its own slot within the second (tools/timer_slots.py) by MonocyteRegen/Start.txt.

let ref rTick := CompileScript "MonocyteRegen/Tick.txt"
let ref rTrue := CompileScript "MonocyteRegen/Always.txt"
let ref rStart := CompileScript "MonocyteRegen/Start.txt"
if eval (IsFormValid rTick) && (IsFormValid rTrue) && (IsFormValid rStart)
	CallAfterSeconds 0.2 rStart 1 rTick rTrue 1 20	; timer slot 0.200 s (tools/timer_slots.py)
else
	print "Monocyte Limb Regen: its script functions failed to compile, so limbs don't regenerate. Details: nvse.log"
endif
