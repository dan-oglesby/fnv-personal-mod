; Target Readout: default settings, set once per save (JIP auxiliary variables on the player, kept
; in JIP's co-save). Change them from the console, e.g. player.AuxVarSetFlt "_trRange" 5000
; Each version's block runs once, in rising order, so older saves get only what was added since.
begin Function {}
	if eval (player.AuxVarGetFlt "_trInit") < 1
		; version 1 (2026-11-17): a new save, or the first load with this mod
		player.AuxVarSetFlt "_trInit" 1
		player.AuxVarSetFlt "_trOn" 1		; the readout is on (0 hides it)
		player.AuxVarSetFlt "_trRange" 3000	; actors up to 3000 units away (about 43 m)
		player.AuxVarSetFlt "_trRed" 1		; text colour: 1 red, 0 the HUD colour, 2 the HUD's alternate colour
	endif
	if eval (player.AuxVarGetFlt "_trInit") < 2
		; version 2 (2026-11-17): the readout types itself out
		player.AuxVarSetFlt "_trInit" 2
		player.AuxVarSetFlt "_trType" 1		; 1 types each new target's readout out, 0 shows it at once
		player.AuxVarSetFlt "_trStep" 3		; letters typed per frame (higher is faster)
		player.AuxVarSetFlt "_trSound" 1	; the terminal's character clicks while it types (0 silent)
	endif
	if eval (player.AuxVarGetFlt "_trInit") < 3
		; version 3 (2026-11-17): only in VATS, so it isn't always lighting up the HUD
		player.AuxVarSetFlt "_trInit" 3
		player.AuxVarSetFlt "_trVats" 1		; 1 only while targeting in VATS, 0 whenever the crosshair is on someone
	endif
	if eval (player.AuxVarGetFlt "_trInit") < 4
		; version 4 (2026-11-17): in VATS the readout sits beside the selected target
		player.AuxVarSetFlt "_trInit" 4
		player.AuxVarSetFlt "_trAimZ" 90		; the point it sits beside: units above the target's feet
	endif
	if eval (player.AuxVarGetFlt "_trInit") < 5
		; version 5 (2026-11-18): the passive scan (Scan.txt)
		player.AuxVarSetFlt "_trInit" 5
		player.AuxVarSetFlt "_trScan" 1		; outline the people around you now and then (0 off)
		player.AuxVarSetFlt "_trScanT" 6		; seconds between scans
		player.AuxVarSetFlt "_trScanD" 2		; seconds an outline lasts
		player.AuxVarSetFlt "_trScanR" 4000	; range in units (about 57 m)
		player.AuxVarSetFlt "_trScanW" 1		; 1 only while your weapon is out, 0 always
		player.AuxVarSetFlt "_trScanLOS" 1	; 1 only those in your line of sight, 0 through walls too
	endif
end
