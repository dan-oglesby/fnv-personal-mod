; Perk Bank (2026-11-15): lets you finish the level-up menu without taking a perk, and saves
; the perk for later. Banked perks come back at your next perk level: after you take that level's
; perk the menu opens again, once per banked perk. Useful when every perk you qualify for is taken
; (the perk page then shows only greyed-out perks and Done can never be clicked).
;   player.AuxVarGetFlt "_pbBank"   how many perks are banked (console: print it)
; Run by JIP LN NVSE's script runner (ln_: new game and every load). Needs xNVSE and JIP LN.
; Re-registering on every load is safe: JIP keeps menu handlers in a set.

let int iBad := 0
let ref rChk := 0
let rChk := CompileScript "PerkBank/InMenu.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "Perk Bank: PerkBank/InMenu.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
let rChk := CompileScript "PerkBank/OnClose.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "Perk Bank: PerkBank/OnClose.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
let rChk := CompileScript "PerkBank/OnOpen.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "Perk Bank: PerkBank/OnOpen.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
let rChk := CompileScript "PerkBank/Spend.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "Perk Bank: PerkBank/Spend.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
let rChk := CompileScript "PerkBank/Watch.txt"
if eval (IsFormValid rChk) == 0
	let iBad += 1
	print "Perk Bank: PerkBank/Watch.txt failed to compile, so the part that uses it is off (details: nvse.log)"
endif
if iBad > 0
	print "Perk Bank: " + $iBad + " script function(s) failed to compile (named above). Everything else in it is running."
	MessageEx "Perk Bank: part of it is off, a script failed to compile (see the console)"
endif
if iBad >= 0	; always (2026-11-17): each part below is registered only if its own function compiled
	player.AuxVarSetFlt "*_pbSpending" 0
	let ref rOpen := CompileScript "PerkBank/OnOpen.txt"
	let ref rClose := CompileScript "PerkBank/OnClose.txt"
	if eval (IsFormValid rOpen)
		SetOnMenuOpenEventHandler rOpen 1 1027
	endif
	if eval (IsFormValid rClose)
		SetOnMenuCloseEventHandler rClose 1 1027
	endif
endif
