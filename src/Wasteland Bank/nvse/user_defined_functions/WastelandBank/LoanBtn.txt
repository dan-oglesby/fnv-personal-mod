; Wasteland Bank: loan buttons. With no loan: 0-3 borrow 100 / 1,000 / 5,000 / as much as your
; credit allows (Credit.txt), due 30 days later. With a loan: 0-3 repay 100 / 500 / 1,000 / all you
; can, from the caps you carry. 4 back. Paying a loan off by its due day raises your credit 5.
int iBtn
int iAmt
int iCarry
int iOwed
int iMax
int iToday
float fLoan
float fC
ref rCaps
ref rNext
ref rFn

begin Function {iBtn}
	if iBtn == 4
		let rNext := CompileScript "WastelandBank/Courier.txt"
		call rNext
		return
	endif
	let rCaps := BuildRef 0 15	; Caps001
	let fLoan := player.AuxVarGetFlt "_wbLoan"
	let iToday := Floor GameDaysPassed
	if fLoan <= 0
		let rFn := CompileScript "WastelandBank/Credit.txt"
		let iMax := call rFn 1
		let iAmt := 100
		if iBtn == 1
			let iAmt := 1000
		elseif iBtn == 2
			let iAmt := 5000
		elseif iBtn == 3
			let iAmt := iMax
		endif
		if iAmt > iMax
			let iAmt := iMax
		endif
		if iAmt < 1
			MessageEx "With your credit, the bank won't lend to you. Pay your way for a while."
		else
			player.AuxVarSetFlt "_wbLoan" iAmt
			player.AuxVarSetFlt "_wbLoanDays" 0
			let iToday += 30
			player.AuxVarSetFlt "_wbLoanDue" iToday
			player.AddItem rCaps iAmt 1
			print "Wasteland Bank: borrowed " + $iAmt + " caps"
		endif
	else
		let iCarry := player.GetItemCount rCaps
		let iOwed := Ceil fLoan
		let iAmt := iOwed
		if iBtn == 0
			let iAmt := 100
		elseif iBtn == 1
			let iAmt := 500
		elseif iBtn == 2
			let iAmt := 1000
		endif
		if iAmt > iOwed
			let iAmt := iOwed
		endif
		if iAmt > iCarry
			let iAmt := iCarry
		endif
		if iAmt > 0
			player.RemoveItem rCaps iAmt 1
			let fLoan -= iAmt
			if fLoan < 0.5
				let fLoan := 0
				player.AuxVarSetFlt "_wbLoanDays" 0
				if eval iToday <= (player.AuxVarGetFlt "_wbLoanDue")
					let fC := player.AuxVarGetFlt "_wbCredit"
					let fC += 5
					if fC > 100
						let fC := 100
					endif
					player.AuxVarSetFlt "_wbCredit" fC
					MessageEx "Loan repaid on time: your credit improves."
				endif
			endif
			player.AuxVarSetFlt "_wbLoan" fLoan
			print "Wasteland Bank: repaid " + $iAmt + " caps"
		endif
	endif
	let rNext := CompileScript "WastelandBank/Loan.txt"
	call rNext
end
