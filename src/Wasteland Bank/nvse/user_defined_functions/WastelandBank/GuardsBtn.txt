; Wasteland Bank: hired-guard buttons at the Caravan Agent (0 hire one, 1 dismiss all, 2 back).
; A new guard takes the first free slot (free or dead), steps out beside the agent and follows you.
int iBtn
int iK
int iSlot
int iOn
int iPlug
int iId
float fSign
float fBal
ref rBase
ref rRef
ref rAgent
ref rFn

begin Function {iBtn}
	if iBtn == 2
		let rFn := CompileScript "WastelandBank/AgentMenu.txt"
		call rFn
		return
	endif
	if eval (player.AuxVarGetFlt "_wgInit") == 0
		let iK := 0
		while iK < 3
			player.AuxVarSetFlt "_wgOn" 0 iK
			player.AuxVarSetFlt "_wgWait" 0 iK
			let iK += 1
		loop
		player.AuxVarSetFlt "_wgInit" 1
	endif
	if iBtn == 1
		let rFn := CompileScript "WastelandBank/GuardGo.txt"
		let iK := 0
		while iK < 3
			if eval (player.AuxVarGetFlt "_wgOn" iK) == 1
				call rFn iK
			endif
			let iK += 1
		loop
		MessageEx "Your hired guards are dismissed."
		return
	endif
	let iSlot := -1
	let iK := 0
	while iK < 3
		let iOn := player.AuxVarGetFlt "_wgOn" iK
		if iSlot < 0 && (iOn == 0 || iOn == 3)
			let iSlot := iK
		endif
		let iK += 1
	loop
	if iSlot < 0
		MessageEx "Three guards is all the company will spare."
		return
	endif
	let fSign := player.AuxVarGetFlt "_wgSign"
	let fBal := player.AuxVarGetFlt "_wbBal"
	if fSign > fBal
		MessageEx "Not enough in the bank to sign a guard on."
		return
	endif
	let rAgent := player.AuxVarGetRef "_wbRepCv"
	if eval (IsFormValid rAgent) == 0
		let rAgent := player
	endif
	let iPlug := GetModIndex "WastelandBank.esp"
	let iId := 2071	; WBGuardA 0x817
	if eval (GetRandomPercent) < 30
		let iId := 2072	; WBGuardB 0x818
	endif
	let rBase := BuildRef iPlug iId
	if eval (IsFormValid rBase) == 0
		return
	endif
	let fBal -= fSign
	player.AuxVarSetFlt "_wbBal" fBal
	let rRef := rAgent.PlaceAtMe rBase 1 128 1
	rRef.SetPlayerTeammate 1
	rRef.EvaluatePackage
	player.AuxVarSetRef "_wgRef" rRef iSlot
	player.AuxVarSetFlt "_wgOn" 1 iSlot
	player.AuxVarSetFlt "_wgWait" 0 iSlot
	let rFn := CompileScript "WastelandBank/GuardAct.txt"
	SetEventHandler "OnActivate" rFn "first"::rRef
	SetEventHandler "OnActivate" rFn "second"::rRef
	MessageEx "A company guard signs on and falls in behind you."
	let rFn := CompileScript "WastelandBank/Guards.txt"
	call rFn
end
