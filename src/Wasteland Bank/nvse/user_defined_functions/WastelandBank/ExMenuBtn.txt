; Wasteland Exchange: main menu buttons (0 New Vegas Exchange, 1 Hub Exchange, 2 funds and the
; bank (board 3), 3 portfolio, 4 news, 5 exchange settings, 6 back to the bank).
int iBtn
ref rNext

begin Function {iBtn}
	if iBtn == 0 || iBtn == 1
		let rNext := CompileScript "WastelandBank/ExBoard.txt"
		call rNext iBtn
	elseif iBtn == 2
		let rNext := CompileScript "WastelandBank/ExBoard.txt"
		call rNext 3
	elseif iBtn == 3
		let rNext := CompileScript "WastelandBank/ExPortfolio.txt"
		call rNext
	elseif iBtn == 4
		let rNext := CompileScript "WastelandBank/ExNewsView.txt"
		call rNext
	elseif iBtn == 5
		let rNext := CompileScript "WastelandBank/ExSettings.txt"
		call rNext
	else
		let rNext := CompileScript "WastelandBank/Open.txt"
		call rNext
	endif
end
