; Wasteland Exchange: the exchange's main menu (from the bank's main menu).
float fBal
float fLong
float fShort
float fDiv
float fNcr
float fLeg
float fNcrCh
float fLegCh
float fPy
ref rPrice
int iN
int iSlot
string_var sHead
string_var sAuto
float fReserve
int iMode
int iId
ref rHead
ref rTot
ref rCb

begin Function {}
	let rTot := CompileScript "WastelandBank/ExTotals.txt"
	let fLong := call rTot 1
	let fShort := call rTot 2
	let fBal := player.AuxVarGetFlt "_wbBal"
	let fDiv := player.AuxVarGetFlt "_wxDiv"
	let sHead := "No market news yet."
	let iN := player.AuxVarGetFlt "_wxNewsN"
	if iN > 0
		let iSlot := (iN - 1) % 10
		let iId := player.AuxVarGetFlt "_wxNewsId" iSlot
		let rHead := CompileScript "WastelandBank/ExHeadline.txt"
		let sHead := call rHead iId
	endif
	; exchange rates: caps per NCR dollar and per denarius, and the change since yesterday
	let rPrice := CompileScript "WastelandBank/ExPrice.txt"
	let fNcr := call rPrice 14
	let fPy := player.AuxVarGetFlt "_wxPy" 14
	let fNcrCh := 0
	if fPy > 0
		let fNcrCh := (fNcr / fPy - 1) * 100
	endif
	let fLeg := call rPrice 15
	let fPy := player.AuxVarGetFlt "_wxPy" 15
	let fLegCh := 0
	if fPy > 0
		let fLegCh := (fLeg / fPy - 1) * 100
	endif
	let fNcr := fNcr * 0.4
	let fLeg := fLeg * 4
	let iMode := player.AuxVarGetFlt "_wxAutoMode"
	let fReserve := player.AuxVarGetFlt "_wxReserve"
	let sAuto := "off"
	if iMode == 1
		let sAuto := "index"
	elseif iMode == 2
		let sAuto := "my stocks"
	endif
	let rCb := CompileScript "WastelandBank/ExMenuBtn.txt"
	MessageBoxExAlt rCb "^Wasteland Exchange^Bank balance (cash): %.0f caps%rShares held: %.0f caps%rShort positions: %.0f caps to cover%rDividends earned: %.0f caps%r%rNCR dollar: %.3f caps (%.1f%% today). Denarius: %.2f caps (%.1f%% today).%rAuto-invest: %z (reserve %.0f caps)%r%rLatest: %z|New Vegas Exchange (caps)|Hub Exchange (NCR dollars)|My portfolio|Market news|Exchange settings|Back to the bank" fBal fLong fShort fDiv fNcr fNcrCh fLeg fLegCh sAuto fReserve sHead
end
