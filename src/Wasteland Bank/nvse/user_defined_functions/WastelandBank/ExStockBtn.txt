; Wasteland Exchange: stock-page buttons (0-2 buy 100 / 1,000 / 10,000 caps, 3 sell half, 4 sell
; all, 5 short 1,000 caps, 6 cover the short, 7 toggle dividend reinvestment, 8 back). Every
; button except Back reopens the stock page.
int iBtn
int iT
int iEx
ref rTrade
ref rNext

begin Function {iBtn}
	let iT := player.AuxVarGetFlt "_wxCurT"
	let rTrade := CompileScript "WastelandBank/ExTrade.txt"
	if iBtn == 8
		let iEx := player.AuxVarGetFlt "_wxCurEx"
		let rNext := CompileScript "WastelandBank/ExBoard.txt"
		call rNext iEx
		return
	endif
	if iBtn == 0
		call rTrade iT 1 100
	elseif iBtn == 1
		call rTrade iT 1 1000
	elseif iBtn == 2
		call rTrade iT 1 10000
	elseif iBtn == 3
		call rTrade iT 2 0.5
	elseif iBtn == 4
		call rTrade iT 2 1
	elseif iBtn == 5
		call rTrade iT 3 1000
	elseif iBtn == 6
		call rTrade iT 4 0
	elseif iBtn == 7
		if eval (player.AuxVarGetFlt "_wxD" iT) == 1
			player.AuxVarSetFlt "_wxD" 0 iT
		else
			player.AuxVarSetFlt "_wxD" 1 iT
		endif
	endif
	let rNext := CompileScript "WastelandBank/ExStock.txt"
	call rNext iT
end
