; Wasteland Bank: a hired guard's orders ("*_wgSel" the slot): wait here or follow, dismiss, close.
int iSlot
int iN
int iK
float fWage
float fDay
string_var sMove
ref rCb

begin Function {}
	let iSlot := player.AuxVarGetFlt "*_wgSel"
	let sMove := "Wait here"
	if eval (player.AuxVarGetFlt "_wgWait" iSlot) == 1
		let sMove := "Follow me"
	endif
	let iN := 0
	let iK := 0
	while iK < 3
		if eval (player.AuxVarGetFlt "_wgOn" iK) == 1
			let iN += 1
			endif
		let iK += 1
	loop
	let fWage := player.AuxVarGetFlt "_wgWage"
	let fDay := fWage * iN
	let rCb := CompileScript "WastelandBank/GuardBtn.txt"
	MessageBoxExAlt rCb "^Hired guard^Paid %.0f caps a day, like each of your %.0f guards (%.0f a day in all), from your bank balance.|%z|Dismiss|Close" fWage iN fDay sMove
end
