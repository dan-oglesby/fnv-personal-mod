; Wasteland Bank: hired guards' wages for iDays game days (from Tick.txt's daily bookkeeping;
; 2026-11-17): each serving guard's "_wgWage" a day, from your bank balance. If the balance can't
; cover it, nothing is paid and they warn you ("_wgUnpaid" days owed); a second unpaid day and
; they all walk off (GuardGo.txt).
int iDays
int iK
int iN
int iOwed
float fCost
float fBal
ref rFn

begin Function {iDays}
	if eval (player.AuxVarGetFlt "_wgInit") == 0
		return
	endif
	if iDays < 1
		return
	endif
	let iN := 0
	let iK := 0
	while iK < 3
		if eval (player.AuxVarGetFlt "_wgOn" iK) == 1
			let iN += 1
			endif
		let iK += 1
	loop
	if iN == 0
		player.AuxVarSetFlt "_wgUnpaid" 0
		return
	endif
	let fCost := player.AuxVarGetFlt "_wgWage"
	let fCost *= iN * iDays
	let fBal := player.AuxVarGetFlt "_wbBal"
	if fBal >= fCost
		let fBal -= fCost
		player.AuxVarSetFlt "_wbBal" fBal
		player.AuxVarSetFlt "_wgUnpaid" 0
		return
	endif
	let iOwed := player.AuxVarGetFlt "_wgUnpaid"
	let iOwed += iDays
	player.AuxVarSetFlt "_wgUnpaid" iOwed
	if iOwed < 2
		MessageBoxEx "Hired guards%rYour guards weren't paid today: %.0f caps, and your balance can't cover it. Pay them by tomorrow or they walk off." fCost
		return
	endif
	let rFn := CompileScript "WastelandBank/GuardGo.txt"
	let iK := 0
	while iK < 3
		if eval (player.AuxVarGetFlt "_wgOn" iK) == 1
			call rFn iK
			endif
		let iK += 1
	loop
	player.AuxVarSetFlt "_wgUnpaid" 0
	MessageBoxEx "Hired guards%rUnpaid two days running, your hired guards walk off."
end
