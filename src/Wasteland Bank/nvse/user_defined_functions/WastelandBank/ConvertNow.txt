; Wasteland Bank: convert NCR dollars, Legion coins and pre-war money in your inventory into caps.
; NCR and Legion money convert at today's exchange rate: the vanilla value times that currency's
; floating multiplier (Wasteland Exchange tickers 14 NCRD and 15 LEGD), exactly, not the item
; value rounded to whole caps. Pre-war money is fixed at 10. Caravan money (the CMNY versions of
; the same notes and coins) converts at its matching item's rate.
; Argument: 1 = quiet (auto-convert), 0 = show a note.
int iQuiet
array_var aItems
array_var aCaravan
array_var aBase
array_var aKind
int i
int iCount
int iTotal
float fValue
float fTotal
float fNcr
float fLeg
ref rItem
ref rCaravan
ref rCaps
ref rPrice

begin Function {iQuiet}
	; FalloutNV.esm form IDs (decimal): Aureus, Denarius, Denarius (mangled), $100, $20, $5 NCR,
	; pre-war money; then the caravan-money twin of each (0 = none); vanilla values; currency
	; (1 Legion, 0 NCR, 2 fixed)
	let aItems := Ar_List 1287385, 1276829, 1520182, 1287384, 1287383, 1287382, 213095
	let aCaravan := Ar_List 1534663, 1534660, 1534661, 1534665, 1534664, 1534662, 0
	let aBase := Ar_List 100, 4, 3, 40, 8, 2, 10
	let aKind := Ar_List 1, 1, 1, 0, 0, 0, 2
	let fNcr := 1
	let fLeg := 1
	if eval (player.AuxVarGetFlt "_wxInit") == 1
		let rPrice := CompileScript "WastelandBank/ExPrice.txt"
		let fNcr := call rPrice 14
		let fLeg := call rPrice 15
	endif
	let fTotal := 0
	let i := 0
	while i < Ar_Size aItems
		let rItem := BuildRef 0 aItems[i]
		let fValue := aBase[i]
		if aKind[i] == 0
			let fValue *= fNcr
		elseif aKind[i] == 1
			let fValue *= fLeg
		endif
		let iCount := player.GetItemCount rItem
		if iCount > 0
			player.RemoveItem rItem iCount 1
			let fTotal += iCount * fValue
		endif
		if aCaravan[i] > 0
			let rCaravan := BuildRef 0 aCaravan[i]
			let iCount := player.GetItemCount rCaravan
			if iCount > 0
				player.RemoveItem rCaravan iCount 1
				let fTotal += iCount * fValue
			endif
		endif
		let i += 1
	loop
	let iTotal := fTotal + 0.5
	if iTotal > 0
		let rCaps := BuildRef 0 15	; Caps001
		player.AddItem rCaps iTotal 1
		print "Wasteland Bank: converted currency into " + $iTotal + " caps"
		if iQuiet == 0
			MessageEx "Converted your currency into %.0f caps." iTotal
		endif
	elseif iQuiet == 0
		MessageEx "No NCR, Legion or pre-war money to convert."
	endif
	SetFunctionValue iTotal
end
