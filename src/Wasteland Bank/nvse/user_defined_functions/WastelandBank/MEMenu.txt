; Wasteland Bank: the goods side of a bank counter: any mailbox or Mojave Express dropbox (PostNear.txt;
; 2026-11-18, before then only the five dropboxes: Goodsprings, Primm, Novac, Freeside, the Vegas
; Boulevard station). Opened from the counter (Post.txt). Away from a counter it says where to go;
; courier delivery away from a counter went with the squads (MEAwayBtn.txt is no longer offered).
;   the vault (Vault.txt), ammo and supplies (Ammo.txt), selling (Sell.txt), the catalogue
;   (Catalogue.txt). Vault traffic is charged by weight, loose ammo is reboxed for 5 percent.
int iNear
int iPost
float fRate
string_var sAt
ref rFn
ref rCb

begin Function {}
	let rFn := CompileScript "WastelandBank/PostNear.txt"
	let iPost := call rFn
	if iPost == 0
		MessageBoxEx "Wasteland Bank%rYour goods (the vault, ammo and supplies, selling, the catalogue) are handled at any bank counter: open a mailbox or a Mojave Express dropbox."
		return
	endif
	let rFn := CompileScript "WastelandBank/MENear.txt"
	let iNear := call rFn
	if iNear < 0
		let sAt := "the mailbox"
	elseif iNear == 0
		let sAt := "Goodsprings"
	elseif iNear == 1
		let sAt := "Primm"
	elseif iNear == 2
		let sAt := "Novac"
	elseif iNear == 3
		let sAt := "Freeside"
	else
		let sAt := "the Vegas Boulevard station"
	endif
	let rFn := CompileScript "WastelandBank/Rate.txt"
	let fRate := call rFn 1
	let rCb := CompileScript "WastelandBank/MEMenuBtn.txt"
	MessageBoxExAlt rCb "^Bank counter: goods (%z)^Carrying to and from the bank costs %.2f caps a pound each way.|Vault storage|Ammo and supplies|Sell things|Order from the catalogue|Repairs|Back" sAt fRate
end
