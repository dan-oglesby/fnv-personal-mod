; Wasteland Bank: withdraw buttons (0 100, 1 500, 2 1000, 3 all, 4 back). Never more than the
; whole caps in the balance, less the courier fee ("_wbFeePct" percent, charged after through
; Fee.txt); interest fractions stay in the bank.
int iBtn
int iAmt
int iWhole
float fBal
ref rCaps
ref rNext
float fFee
float fPct

begin Function {iBtn}
	if iBtn == 4
		let rNext := CompileScript "WastelandBank/Courier.txt"
		call rNext
		return
	endif
	let fBal := player.AuxVarGetFlt "_wbBal"
	let fPct := player.AuxVarGetFlt "_wbFeePct"
	let fBal := fBal / (1 + fPct / 100)
	let iWhole := Floor fBal
	let iAmt := iWhole
	if iBtn == 0
		let iAmt := 100
	elseif iBtn == 1
		let iAmt := 500
	elseif iBtn == 2
		let iAmt := 1000
	endif
	if iAmt > iWhole
		let iAmt := iWhole
	endif
	if iAmt > 0
		let rCaps := BuildRef 0 15	; Caps001
		let fBal := player.AuxVarGetFlt "_wbBal"
		let fBal -= iAmt
		player.AuxVarSetFlt "_wbBal" fBal
		player.AddItem rCaps iAmt 1
		print "Wasteland Bank: withdrew " + $iAmt + " caps"
		let fFee := iAmt * fPct / 100
		let rNext := CompileScript "WastelandBank/Fee.txt"
		call rNext fFee
	endif
	let rNext := CompileScript "WastelandBank/Withdraw.txt"
	call rNext
end
