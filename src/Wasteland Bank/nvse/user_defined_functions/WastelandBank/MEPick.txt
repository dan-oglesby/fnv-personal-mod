; Wasteland Bank: visiting a Mojave Express dropbox, checked every second (SqTick.txt). Walking up to
; one (within about 6 m: MENear.txt), once per visit ("*_wbMEVisit", cleared when you walk away):
; - your top-ups from the vault, per the Ammo and supplies settings: this ammo to 300, a magazine for
;   weapons with none, N of each aid item (BankAmmo.txt 8; moved here from the couriers 2026-11-17);
; - the first time ever, a note that your bank card opens the dropbox's services (MEMenu.txt).
; And any Mojave Express order that's ready ("_wbShipAt" passed) is handed over (WBInTransitRef
; 0x80E). iForce is no longer used (orders came with squads before 2026-11-17).
int iForce
int iPlug
int iNear
int iAmt
float fNow
ref rShip
ref rFn

begin Function {iForce}
	let rFn := CompileScript "WastelandBank/MENear.txt"
	let iNear := call rFn
	if iNear < 0
		player.AuxVarSetFlt "*_wbMEVisit" 0
		return
	endif
	if eval (player.AuxVarGetFlt "*_wbMEVisit") == 0
		player.AuxVarSetFlt "*_wbMEVisit" 1
		if eval (player.AuxVarGetFlt "_wbMESeen") == 0
			player.AuxVarSetFlt "_wbMESeen" 1
			MessageBoxEx "Mojave Express%rYour bank's goods go through these dropboxes: the vault, ammo and supplies, selling, and catalogue orders. Open them from your bank card's main menu while you're here. Each visit, the dropbox tops up your ammo and aid from the vault."
		endif
		if eval (player.AuxVarGetFlt "_wbArrAmmo") > 0 || (player.AuxVarGetFlt "_wbRefillN") > 0
			let rFn := CompileScript "WastelandBank/BankAmmo.txt"
			let iAmt := call rFn 8
			if iAmt > 0
				MessageEx "Mojave Express topped you up from the vault: %.0f rounds and items." iAmt
			endif
		endif
	endif
	let iPlug := GetModIndex "WastelandBank.esp"
	let rShip := BuildRef iPlug 2062	; WBInTransitRef
	if eval (IsFormValid rShip) == 0
		return
	endif
	if eval (rShip.GetNumItems) == 0
		return
	endif
	let fNow := GameDaysPassed
	if eval fNow < (player.AuxVarGetFlt "_wbShipAt")
		return
	endif
	rShip.RemoveAllItems player 1
	MessageEx "You collect your Mojave Express order from the dropbox."
end
