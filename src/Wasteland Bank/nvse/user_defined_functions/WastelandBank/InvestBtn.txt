; Wasteland Bank: investment buttons. Invest.txt lists only the services your stage has opened and
; notes which service each button is ("*_wbInvC": 0 exchange, 1 war bonds, 2 currency accounts,
; 3 bank notes, 4 managed account, 5 Legion treasury, 6 running the bank, 7 holdings by sector,
; 8 operations, 9 back).
int iBtn
int iCode
ref rNext

begin Function {iBtn}
	let iCode := player.AuxVarGetFlt "*_wbInvC" iBtn
	if iCode == 0
		let rNext := CompileScript "WastelandBank/ExMenu.txt"
	elseif iCode == 1
		let rNext := CompileScript "WastelandBank/Bonds.txt"
	elseif iCode == 2
		let rNext := CompileScript "WastelandBank/Accounts.txt"
	elseif iCode == 3
		let rNext := CompileScript "WastelandBank/BankNotes.txt"
	elseif iCode == 4
		let rNext := CompileScript "WastelandBank/Managed.txt"
	elseif iCode == 5
		let rNext := CompileScript "WastelandBank/Legion.txt"
	elseif iCode == 6
		let rNext := CompileScript "WastelandBank/OwnerFees.txt"
	elseif iCode == 7
		let rNext := CompileScript "WastelandBank/Sectors.txt"
	elseif iCode == 8
		let rNext := CompileScript "WastelandBank/Ops.txt"
	else
		let rNext := CompileScript "WastelandBank/Courier.txt"
	endif
	call rNext
end
