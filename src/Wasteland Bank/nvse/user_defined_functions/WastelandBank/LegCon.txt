; Wasteland Bank: publicani contracts, Rome's private contractors for the state. Each is sold once
; and runs 30 days; it pays a share every week (4 percent of its price, times how the frontier is
; going) and returns its price at the end. Offered again 60 days after it was bought.
;   0 Grain for the Fort, 200 denarii: pays double once the Legion marches on the Dam (event 29).
;   1 Arizona tribute, 400 denarii: pays 1.25 times.
;   2 The caravan roads, 300 denarii: pays half once the Khans break with Caesar (event 19).
int iK
int iToday
float fDay
string_var s0
string_var s1
string_var s2
string_var sOne
ref rCb

begin Function {}
	let iToday := Floor GameDaysPassed
	let iK := 0
	while iK < 3
		if eval (player.AuxVarGetFlt "_ltC" iK) > 0
			let fDay := player.AuxVarGetFlt "_ltCEnd" iK
			let sOne := sv_Construct "yours until day %.0f" fDay
		elseif eval (player.AuxVarGetFlt "_ltCNext" iK) > iToday
			let fDay := player.AuxVarGetFlt "_ltCNext" iK
			let sOne := sv_Construct "offered again from day %.0f" fDay
		else
			let sOne := "on offer"
		endif
		if iK == 0
			let s0 := sOne
		elseif iK == 1
			let s1 := sOne
		else
			let s2 := sOne
		endif
		let iK += 1
	loop
	let rCb := CompileScript "WastelandBank/LegConBtn.txt"
	MessageBoxExAlt rCb "^Publicani contracts^Today is day %.0f.%rGrain for the Fort (200 denarii): %z%rArizona tribute (400 denarii): %z%rThe caravan roads (300 denarii): %z|Buy Grain for the Fort|Buy Arizona tribute|Buy The caravan roads|Back" iToday s0 s1 s2
end
