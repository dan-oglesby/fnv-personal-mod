; Wasteland Bank: caravan ventures (2026-11-17), arranged in person with a representative
; (RepAct.txt sets "*_cvSide"):
;   0 the Caravan Agent at the Crimson Caravan Company, from the Investor stage: routes 0-4, in caps;
;   1 the Publicanus at Cottonwood Cove, the Legion's caravans, from the Financier stage, for those
;     the Legion accepts (LegRep.txt) while its treasury stands: routes 5-6, in denarii from your
;     denarius account. (They replace the publicani contracts.)
; Back a run with a large stake (CvStake.txt). It sets out 6 hours later from its starting point;
; be there within a day to escort it yourself (it walks the road with you, and may be ambushed): get
; it to its destination and it pays a high return, for certain. Otherwise it travels on its own and
; the route's odds decide (CvTick.txt, CvEnd.txt). One venture at a time, with either.
;   on its own: succeeds 1 - 0.7 x danger of the time, paying 15 percent + 50 x danger; fails and
;   only 30 percent is salvaged. Escorted to the end: 60 percent + 60 x danger. Danger: CvDanger.txt.
int iSide
int iR
int iFirst
int iLast
int iState
int iOk
float fD
float fP
float fWin
float fEsc
float fStake
float fNow
float fH
float fDpct
string_var sName
string_var sWho
string_var sCur
string_var sBody
string_var sBtns
ref rFn
ref rName
ref rCb

begin Function {}
	let iSide := player.AuxVarGetFlt "*_cvSide"
	let sWho := "Caravan Agent"
	let iFirst := 0
	let iLast := 4
	if iSide == 1
		let sWho := "Publicanus"
		let iFirst := 5
		let iLast := 6
	endif
	let rFn := CompileScript "WastelandBank/Stage.txt"
	if iSide == 0
		let iOk := call rFn 2 1
		if iOk == 0
			MessageBoxEx "Caravan Agent%rThe Crimson Caravan Company takes outside money only from Wasteland Bank's investors. Come back when the bank counts you one: the Investor stage, at 5,000 caps of wealth."
			return
		endif
	else
		let iOk := call rFn 4 1
		if iOk == 0
			MessageBoxEx "Publicanus%rThe Legion lets its caravans to men of substance. Come back when Wasteland Bank counts you a Financier, at 75,000 caps of wealth."
			return
		endif
		if eval (player.AuxVarGetFlt "_ltDefault") == 1 || (player.AuxVarGetFlt "_ltClosed") == 1
			MessageBoxEx "Publicanus%rThe Dam is lost and the treasury has defaulted. No caravans leave the Cove on Rome's account now."
			return
		endif
		let rFn := CompileScript "WastelandBank/LegRep.txt"
		let iOk := call rFn
		if iOk != 1
			MessageBoxEx "Publicanus%rThe Legion does not trust its roads to you. Come back when it counts you at least Accepted."
			return
		endif
	endif

	let iState := player.AuxVarGetFlt "_cvState"
	let rName := CompileScript "WastelandBank/CvName.txt"
	let rCb := CompileScript "WastelandBank/CaravansBtn.txt"
	if iState > 0
		let iR := player.AuxVarGetFlt "_cvRoute"
		let sName := call rName iR
		let fStake := player.AuxVarGetFlt "_cvStake"
		let sCur := "caps"
		if eval (player.AuxVarGetFlt "_cvCur") == 1
			let sCur := "denarii"
		endif
		let fNow := GameDaysPassed
		if iState == 1
			let fH := player.AuxVarGetFlt "_cvDep"
			let fH := (fH - fNow) * 24
			if fH > 0
				let sBody := sv_Construct "Your caravan (%.0f %z), %z, sets out in about %.0f hours. Be at its starting point to escort it." fStake sCur sName fH
			else
				let fH := player.AuxVarGetFlt "_cvArr"
				let fH := (fH - fNow) * 24
				let sBody := sv_Construct "Your caravan (%.0f %z), %z, is on the road without you: it arrives in about %.0f hours, and the route's odds decide." fStake sCur sName fH
			endif
		else
			let sBody := sv_Construct "Your caravan (%.0f %z), %z, is on the road with you. Keep it alive and in sight." fStake sCur sName
		endif
		let sBody := sBody + (sv_Construct "%rOne venture at a time: come back when it's over.")
		MessageBoxExAlt rCb "^%z^%z|Leave" sWho sBody
		return
	endif
	let rFn := CompileScript "WastelandBank/CvDanger.txt"
	if iSide == 1
		let sBody := "Back one of the Legion's caravans, in denarii. Walk it in yourself for a high, certain return, or let it take its chances with the NCR."
	else
		let sBody := "Back a caravan run. Escort it yourself for a high, certain return, or let it take its chances."
	endif
	let sBtns := ""
	let iR := iFirst
	while iR <= iLast
		let fD := call rFn iR
		let fP := (1 - 0.7 * fD) * 100
		let fWin := (0.15 + 0.5 * fD) * 100
		let fEsc := (0.6 + 0.6 * fD) * 100
		let sName := call rName iR
		let fDpct := fD * 100
		let sBody := sBody + (sv_Construct "%r%z: danger %.0f. Alone: %.0f percent chance of +%.0f percent. Escorted: +%.0f percent." sName fDpct fP fWin fEsc)
		let sBtns := sBtns + "|" + sName
		let iR += 1
	loop
	let sBtns := sBtns + "|Leave"
	MessageBoxExAlt rCb "^%z^%z%z" sWho sBody sBtns
end
