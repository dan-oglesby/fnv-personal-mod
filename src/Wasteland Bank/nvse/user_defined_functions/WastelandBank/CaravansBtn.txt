; Wasteland Bank: caravan venture buttons (Caravans.txt). With a venture running there's only
; Leave; otherwise each button is a route of the representative's ("*_cvSide": the Caravan Agent's
; 0-4, the Publicanus's 5-6), then Leave.
int iBtn
int iSide
int iFirst
int iN
ref rFn

begin Function {iBtn}
	let iSide := player.AuxVarGetFlt "*_cvSide"
	let iFirst := 0
	let iN := 5
	if iSide == 1
		let iFirst := 5
		let iN := 2
	endif
	if eval (player.AuxVarGetFlt "_cvState") > 0 || iBtn >= iN
		return
	endif
	let iBtn += iFirst
	player.AuxVarSetFlt "*_cvPick" iBtn
	let rFn := CompileScript "WastelandBank/CvStake.txt"
	call rFn
end
