; Wasteland Bank: Mojave Express buttons (0 vault, 1 ammo and supplies, 2 sell, 3 catalogue,
; 4 repairs (Fix.txt), 5 back to the counter (Post.txt).
int iBtn
ref rNext

begin Function {iBtn}
	if iBtn == 0
		let rNext := CompileScript "WastelandBank/Vault.txt"
	elseif iBtn == 1
		let rNext := CompileScript "WastelandBank/Ammo.txt"
	elseif iBtn == 2
		let rNext := CompileScript "WastelandBank/Sell.txt"
	elseif iBtn == 3
		let rNext := CompileScript "WastelandBank/Catalogue.txt"
	elseif iBtn == 4
		let rNext := CompileScript "WastelandBank/Fix.txt"
	else
		let rNext := CompileScript "WastelandBank/Post.txt"
	endif
	call rNext
end
