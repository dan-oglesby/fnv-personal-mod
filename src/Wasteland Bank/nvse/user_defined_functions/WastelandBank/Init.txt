; Wasteland Bank: first-run defaults, once per character (aux variables on the player).
;   _wbBal     balance in caps            _wbEarned  interest earned so far
;   _wbLoan    loan owed in caps          _wbLoanDays game days counted toward the next loan week
;   _wbDay     last whole game day booked _wbAutoDep / _wbAutoConv  1 = on
;   _wbKeep    caps auto-deposit leaves in your pocket

int iToday

begin Function {}
	if eval (player.AuxVarGetFlt "_wbInit") == 0
		let iToday := Floor GameDaysPassed
		player.AuxVarSetFlt "_wbDay" iToday
		player.AuxVarSetFlt "_wbKeep" 200
		player.AuxVarSetFlt "_wbInit" 1
	endif
end
