; Wasteland Bank: first-run defaults, once per character (aux variables on the player).
;   _wbBal     balance in caps            _wbEarned  interest earned so far
;   _wbLoan    loan owed in caps          _wbLoanDays game days counted toward the next loan week
;   _wbDay     last whole game day booked _wbAutoDep / _wbAutoConv  1 = on
;   _wbKeep    caps auto-deposit leaves in your pocket
;   _wbFeePct  courier fee, percent of the cash moved (2)   _wbVaultRate  vault fee, caps a pound (1)
;   _wbSellCut couriers' cut of what they sell for you, percent (10)
;   _wbSet     settings version; each version's block runs once, in order (2026-11-17: the version-2
;              block used to sit last, after version 3 had already been set, so it never ran and the
;              selling cut stayed 0; version 8 puts it right)

int iToday
int iK
ref rFn

begin Function {}
	if eval (player.AuxVarGetFlt "_wbInit") == 0
		let iToday := Floor GameDaysPassed
		player.AuxVarSetFlt "_wbDay" iToday
		player.AuxVarSetFlt "_wbKeep" 200
		player.AuxVarSetFlt "_wbInit" 1
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 1
		player.AuxVarSetFlt "_wbSet" 1
		player.AuxVarSetFlt "_wbFeePct" 2
		player.AuxVarSetFlt "_wbVaultRate" 1
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 2
		player.AuxVarSetFlt "_wbSet" 2
		player.AuxVarSetFlt "_wbSellCut" 10
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 3
		player.AuxVarSetFlt "_wbSet" 3
		player.AuxVarSetFlt "_wmaRisk" 2
		let iK := 0
		while iK < 16
			player.AuxVarSetFlt "_wmaW" 0 iK
			player.AuxVarSetFlt "_wmaXC" 0 iK
			let iK += 1
		loop
		let iK := 0
		while iK < 3
			player.AuxVarSetFlt "_wmaXF" 0 iK
			let iK += 1
		loop
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 4
		player.AuxVarSetFlt "_wbSet" 4
		let iK := 0
		while iK < 3
			player.AuxVarSetFlt "_wbOpsF" 0 iK
			let iK += 1
		loop
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 5
		player.AuxVarSetFlt "_wbSet" 5
		player.AuxVarSetFlt "_wbArrAmmo" 1
		if eval (player.AuxVarGetFlt "_wbAmmoOrder") == 1
			player.AuxVarSetFlt "_wbArrAmmo" 2
		endif
		player.AuxVarSetFlt "_wbRefillN" 1
		player.AuxVarSetFlt "_wbAmmoTo" 300
		if eval (player.AuxVarGetFlt "_wbStage") < 4 && (player.AuxVarGetFlt "_wmaRisk") > 1
			player.AuxVarSetFlt "_wmaRisk" 1
		endif
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 6
		player.AuxVarSetFlt "_wbSet" 6
		player.AuxVarSetFlt "_wbCardPay" 1	; pay by bank card at merchants (CardPay.txt)
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 7
		player.AuxVarSetFlt "_wbSet" 7
		player.AuxVarSetFlt "_wbCredit" 60	; credit score (Credit.txt): C to start
		let iK := Floor GameDaysPassed
		player.AuxVarSetFlt "_wbCreditWk" iK
		if eval (player.AuxVarGetFlt "_wbLoan") > 0
			let iK += 30
			player.AuxVarSetFlt "_wbLoanDue" iK
		endif
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 8
		player.AuxVarSetFlt "_wbSet" 8
		if eval (player.AuxVarGetFlt "_wbSellCut") <= 0
			player.AuxVarSetFlt "_wbSellCut" 10	; the cut the out-of-order version 2 never set
		endif
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 9
		player.AuxVarSetFlt "_wbSet" 9
		player.AuxVarSetFlt "_wbDelivCut" 25	; courier delivery's commission, percent of the goods' value (CatClose.txt)
	endif
	if eval (player.AuxVarGetFlt "_wbSet") < 10
		player.AuxVarSetFlt "_wbSet" 10
		player.AuxVarSetFlt "_wgSign" 200	; hired guards: signing fee, caps (Guards.txt)
		player.AuxVarSetFlt "_wgWage" 40	; and each one's wage, caps a day (GuardDay.txt)
	endif
	let rFn := CompileScript "WastelandBank/LegInit.txt"
	call rFn
end
