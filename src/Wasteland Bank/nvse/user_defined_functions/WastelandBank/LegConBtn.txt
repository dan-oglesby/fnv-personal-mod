; Wasteland Bank: contract buttons (0-2 buy that contract, 3 back to the treasury).
int iBtn
int iRep
int iToday
int iDay
float fCost
float fAcct
float fLeg
float fV
ref rFn

begin Function {iBtn}
	if iBtn == 3
		let rFn := CompileScript "WastelandBank/Legion.txt"
		call rFn
		return
	endif
	let rFn := CompileScript "WastelandBank/Stage.txt"
	let iRep := call rFn 4 0	; publicani contracts open at the Financier stage
	if iRep == 0
		let rFn := CompileScript "WastelandBank/LegCon.txt"
		call rFn
		return
	endif
	let rFn := CompileScript "WastelandBank/LegRep.txt"
	let iRep := call rFn
	let iToday := Floor GameDaysPassed
	let fCost := 200
	if iBtn == 1
		let fCost := 400
	elseif iBtn == 2
		let fCost := 300
	endif
	let fAcct := player.AuxVarGetFlt "_wbBalL"
	if eval (player.AuxVarGetFlt "_ltDefault") == 1 || (player.AuxVarGetFlt "_ltClosed") == 1
		MessageEx "The Legion offers no more contracts."
	elseif iRep != 1
		MessageEx "The treasury won't deal with you: your standing with the Legion is below Accepted."
	elseif eval (player.AuxVarGetFlt "_ltC" iBtn) > 0 || (player.AuxVarGetFlt "_ltCNext" iBtn) > iToday
		MessageEx "That contract isn't on offer now."
	elseif fCost > fAcct
		MessageEx "Not enough denarii in your account."
	else
		let fAcct -= fCost
		player.AuxVarSetFlt "_wbBalL" fAcct
		player.AuxVarSetFlt "_ltC" fCost iBtn
		let iDay := iToday + 30
		player.AuxVarSetFlt "_ltCEnd" iDay iBtn
		let iDay := iToday + 60
		player.AuxVarSetFlt "_ltCNext" iDay iBtn
		let rFn := CompileScript "WastelandBank/ExPrice.txt"
		let fLeg := call rFn 15
		let fV := fCost * fLeg * 4
		let rFn := CompileScript "WastelandBank/Flow.txt"
		call rFn 15 fV
		MessageEx "Contract bought for %.0f denarii." fCost
	endif
	let rFn := CompileScript "WastelandBank/LegCon.txt"
	call rFn
end
