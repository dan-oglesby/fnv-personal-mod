; Wasteland Bank: once a second of play (ln_wasteland_bank.txt; not in menus or paused).
; Since 2026-11-18 there are no courier squads: every mailbox and Mojave Express dropbox is a bank
; counter (Post.txt). What's left here:
; - Mojave Express visits and orders at a dropbox (MEPick.txt), as before;
; - a squad still out in a save from before the change ("_wbSqState" 2 here, 3 leaving) is sent
;   home at once (SqLeave.txt 1). One called and not yet arrived (1) is dropped, unless it was called
;   with a larger sum: that becomes an order for the sum ("_wbSqState" 4), collected at any counter
;   from when it was due (PostVisit.txt). 4 is left alone.
; slot weight: light (tools/timer_slots.py): five distances and a state check.
int iState
float fNorm
ref rFn
ref rRateFn

begin Function {}
	let rFn := CompileScript "WastelandBank/MEPick.txt"
	call rFn 0
	let iState := player.AuxVarGetFlt "_wbSqState"
	if iState == 0 || iState == 4
		return
	endif
	if iState == 1
		; called before the change: an ordinary call is dropped, a larger sum becomes an order
		let rRateFn := CompileScript "WastelandBank/Rate.txt"
		let fNorm := call rRateFn 3
		if eval (player.AuxVarGetFlt "_wbSqFloat") > fNorm
			player.AuxVarSetFlt "_wbSqState" 4
		else
			player.AuxVarSetFlt "_wbSqState" 0
			player.AuxVarSetFlt "_wbSqFloat" 0
		endif
		return
	endif
	; here or leaving: home at once
	let rFn := CompileScript "WastelandBank/SqLeave.txt"
	call rFn 1
	player.AuxVarSetFlt "_wbSqState" 0
	player.AuxVarSetFlt "_wbSqFloat" 0
end
