; Wasteland Bank: the courier squad, once a second of play (ln_wasteland_bank.txt; not in menus or
; paused). "_wbSqState" 0 none, 1 called (arrives at game time "_wbSqAt"), 2 here (refs
; "_wbSqRef" 0-2, arrived at "_wbSqArr"), 3 leaving (running off to where they came from).
; - Arrival (2026-11-17): once the time comes and you're not in combat or VATS, three couriers
;   (WastelandBank.esp 0x801-0x803, essential copies of the vanilla mercenaries) appear out of
;   sight behind you (about 21 m outdoors, 6 m indoors, at a marker "_wbSqMark", vanilla XMarker)
;   and run up to you (vanilla DefaultFollowPlayerFar 0x1554DC as a script package, JIP
;   SetAlwaysRun); once close they settle near you. If they haven't reached you in a minute (stuck
;   on scenery), they're moved to you.
; - Walking up to them (within about 4 m) the first time opens their services (Courier.txt); after
;   that, activating one does (CourierAct.txt: they have no dialogue), or the bank card while you're
;   within about 15 m (SqHere.txt).
; - On arrival they take care of auto-deposit (caps over "_wbKeep") and auto-convert, if those are
;   on, with the usual fee.
; - They carry "_wbSqFloat" caps of your balance (2,500 for an ordinary call, more if ordered a
;   day ahead), never more than the balance holds: "_wbSqCash", the most you can withdraw.
;   They also bring any cash-out order that is due (Deliver.txt).
; - They leave (SqLeave.txt) when sent off, or after 3 game hours once you're 15 m away: running
;   back the way they came; or at once if you're about 60 m away, out of sight.
; Goods (the vault, ammo, selling, orders) are at the Mojave Express dropboxes now (MEMenu.txt);
; MEPick.txt runs here every second for them.
int iState
int iPlug
int iIdx
int iId
int iT
float fNow
float fD
float fFar
ref rBase
ref rRef
ref rLead
ref rFn
ref rFee
ref rCaps
ref rMark
ref rPkg
int iCarry
int iAmt
float fBal
float fFee
ref rRateFn

begin Function {}
	; Mojave Express: orders, visits, top-ups at a dropbox (MEPick.txt)
	let rFn := CompileScript "WastelandBank/MEPick.txt"
	call rFn 0
	let iState := player.AuxVarGetFlt "_wbSqState"
	if iState == 0
		return
	endif
	if player.GetDead
		return
	endif
	if IsInKillCam
		return
	endif
	let fNow := GameDaysPassed
	let rLead := player.AuxVarGetRef "_wbSqRef" 0
	; a squad already here after a load: set their activation handlers again (the loader clears
	; "*_wbSqH" on every load; review 2026-11-18)
	if eval iState == 2 && (player.AuxVarGetFlt "*_wbSqH") == 0
		player.AuxVarSetFlt "*_wbSqH" 1
		let rFn := CompileScript "WastelandBank/CourierAct.txt"
		let iIdx := 0
		while iIdx < 3
			let rRef := player.AuxVarGetRef "_wbSqRef" iIdx
			let iIdx += 1
			if eval IsFormValid rRef
				RemoveEventHandler "OnActivate" rFn "first"::rRef
				RemoveEventHandler "OnActivate" rFn "second"::rRef
				SetEventHandler "OnActivate" rFn "first"::rRef
				SetEventHandler "OnActivate" rFn "second"::rRef
			endif
		loop
	endif

	; leaving: gone once they're out of sight, or after 45 seconds
	if iState == 3
		let iT := player.AuxVarGetFlt "*_wbSqLeaveT"
		let iT += 1
		player.AuxVarSetFlt "*_wbSqLeaveT" iT
		let fD := 99999
		if eval IsFormValid rLead
			let fD := player.GetDistance rLead
		endif
		if fD > 1500 || iT > 45
			let rFn := CompileScript "WastelandBank/SqLeave.txt"
			call rFn 1
		endif
		return
	endif

	if iState == 1
		if eval fNow < (player.AuxVarGetFlt "_wbSqAt")
			return
		endif
		if player.IsInCombat
			return
		endif
		if eval (IsModLoaded "WastelandBank.esp") == 0
			player.AuxVarSetFlt "_wbSqState" 0
			return
		endif
		let iPlug := GetModIndex "WastelandBank.esp"
		let fFar := 1500
		if eval player.IsInInterior
			let fFar := 400
		endif
		let rBase := BuildRef 0 59	; XMarker
		let rMark := player.PlaceAtMe rBase 1 fFar 1	; behind you, out of sight
		player.AuxVarSetRef "_wbSqMark" rMark
		let rPkg := BuildRef 0 1397980	; DefaultFollowPlayerFar (0x1554DC)
		let iIdx := 0
		while iIdx < 3
			let iId := 2049 + iIdx	; WBCourierA-C 0x801-0x803
			let rBase := BuildRef iPlug iId
			let rRef := rMark.PlaceAtMe rBase 1 64 iIdx
			rRef.AddScriptPackage rPkg
			rRef.SetAlwaysRun 1
			rRef.EvaluatePackage
			player.AuxVarSetRef "_wbSqRef" rRef iIdx
			let rFn := CompileScript "WastelandBank/CourierAct.txt"
			SetEventHandler "OnActivate" rFn "first"::rRef
			SetEventHandler "OnActivate" rFn "second"::rRef
			let iIdx += 1
			player.AuxVarSetFlt "*_wbSqH" 1
		loop
		player.AuxVarSetFlt "_wbSqArr" fNow
		player.AuxVarSetFlt "_wbSqMet" 0
		player.AuxVarSetFlt "*_wbSqComing" 1
		player.AuxVarSetFlt "*_wbSqComeT" 0
		player.AuxVarSetFlt "_wbSqState" 2
		MessageEx "A Wasteland Bank courier squad is coming up behind you."
		let fFee := player.AuxVarGetFlt "_wbSqFloat"
		if fFee <= 0
			let rRateFn := CompileScript "WastelandBank/Rate.txt"
			let fFee := call rRateFn 3
		endif
		let fBal := player.AuxVarGetFlt "_wbBal"
		let fBal := Floor fBal
		if fFee > fBal
			let fFee := fBal
		endif
		player.AuxVarSetFlt "_wbSqCash" fFee
		let rFn := CompileScript "WastelandBank/Deliver.txt"
		call rFn
		let rFee := CompileScript "WastelandBank/Fee.txt"
		if eval (player.AuxVarGetFlt "_wbAutoConv") == 1
			let rFn := CompileScript "WastelandBank/ConvertNow.txt"
			let iAmt := call rFn 1
			if iAmt > 0
				let rRateFn := CompileScript "WastelandBank/Rate.txt"
				let fFee := call rRateFn 0
				let fFee := iAmt * fFee / 100
				call rFee fFee
			endif
		endif
		if eval (player.AuxVarGetFlt "_wbAutoDep") == 1
			let rCaps := BuildRef 0 15	; Caps001
			let iCarry := player.GetItemCount rCaps
			let fFee := player.AuxVarGetFlt "_wbKeep"
			let iAmt := iCarry - fFee
			if iAmt > 0
				player.RemoveItem rCaps iAmt 1
				let fBal := player.AuxVarGetFlt "_wbBal"
				let fBal += iAmt
				player.AuxVarSetFlt "_wbBal" fBal
				let rRateFn := CompileScript "WastelandBank/Rate.txt"
				let fFee := call rRateFn 0
				let fFee := iAmt * fFee / 100
				call rFee fFee
				MessageEx "The couriers took %.0f caps to the bank." iAmt
			endif
		endif
		return
	endif

	; here
	if eval (IsFormValid rLead) == 0
		let rFn := CompileScript "WastelandBank/SqLeave.txt"
		call rFn 1
		return
	endif
	let fD := player.GetDistance rLead
	; running up: settle once close, or be moved to you after a minute
	if eval (player.AuxVarGetFlt "*_wbSqComing") == 1
		let iT := player.AuxVarGetFlt "*_wbSqComeT"
		let iT += 1
		player.AuxVarSetFlt "*_wbSqComeT" iT
		if fD <= 300 || iT >= 60
			player.AuxVarSetFlt "*_wbSqComing" 0
			let rPkg := BuildRef 0 1397980
			let iIdx := 0
			while iIdx < 3
				let rRef := player.AuxVarGetRef "_wbSqRef" iIdx
				let iIdx += 1
				if eval IsFormValid rRef
					if eval (player.GetDistance rRef) > 600
						rRef.MoveTo player 0 128 0
					endif
					rRef.RemoveScriptPackage rPkg
					rRef.SetAlwaysRun 0
					rRef.EvaluatePackage
				endif
			loop
			let fD := player.GetDistance rLead
		endif
		return
	endif
	if fD > 4000
		let rFn := CompileScript "WastelandBank/SqLeave.txt"
		call rFn 1
		return
	endif
	if eval fNow - (player.AuxVarGetFlt "_wbSqArr") > 0.125 && fD > 1000
		let rFn := CompileScript "WastelandBank/SqLeave.txt"
		call rFn 2
		return
	endif
	if fD <= 256
		let rFn := CompileScript "WastelandBank/SqGive.txt"
		call rFn
		if eval (player.AuxVarGetFlt "_wbSqMet") == 0
			player.AuxVarSetFlt "_wbSqMet" 1
			let rFn := CompileScript "WastelandBank/Courier.txt"
			call rFn
		endif
	endif
end
