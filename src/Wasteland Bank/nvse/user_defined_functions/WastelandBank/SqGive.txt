; Wasteland Bank: what a bank counter hands you when you open it (PostVisit.txt; until 2026-11-18 a
; courier squad, when they reached you):
; - a courier delivery that's due ("_wbDelivAt" passed; WBDeliveryRef 0x816, CatClose.txt);
; - this week's Mojave Market Report, if you haven't had it (Report.txt; a week is 7 game days,
;   "_wbRepWk"). A week you don't visit a counter has no report.
int iPlug
int iWk
ref rDel
ref rFn

begin Function {}
	let iPlug := GetModIndex "WastelandBank.esp"
	let rDel := BuildRef iPlug 2070	; WBDeliveryRef 0x816
	if eval IsFormValid rDel
		if eval (rDel.GetNumItems) > 0 && GameDaysPassed >= (player.AuxVarGetFlt "_wbDelivAt")
			rDel.RemoveAllItems player 1
			MessageEx "Your delivery is waiting at the counter: handed over."
		endif
	endif
	let iWk := Floor (GameDaysPassed / 7)
	if eval iWk > (player.AuxVarGetFlt "_wbRepWk")
		let rFn := CompileScript "WastelandBank/Report.txt"
		if eval IsFormValid rFn
			call rFn iWk
		endif
	endif
end
