; Wasteland Bank: order a larger sum than a counter holds (2026-11-18; before, a squad called with
; it). It's ready at any counter from about a day later (PostVisit.txt). One order at a time.
float fBal
float fNow
float fAt
int iHrs
ref rCb

begin Function {}
	if eval (player.AuxVarGetFlt "_wbSqState") == 4
		let fNow := GameDaysPassed
		let fAt := player.AuxVarGetFlt "_wbSqAt"
		let iHrs := (fAt - fNow) * 24 + 0.5
		if iHrs < 1
			MessageEx "Your larger sum is ready: open any bank counter."
		else
			MessageEx "A larger sum is already on its way to the counters: about %.0f hours." iHrs
		endif
		return
	endif
	let fBal := player.AuxVarGetFlt "_wbBal"
	let rCb := CompileScript "WastelandBank/OrderCashBtn.txt"
	MessageBoxExAlt rCb "^Order a larger sum^A counter pays out up to its daily float. A larger sum takes a day to arrange, then any mailbox or Mojave Express dropbox has it.%rBalance: %.0f caps.|5,000 caps|10,000 caps|25,000 caps|My whole balance|Cancel" fBal
end
