; Wasteland Bank: call a courier squad (from the main menu). An ordinary call arrives in 1 to 2 game
; hours carrying a float of 2,500 caps from your balance, the most you can withdraw from it. A
; larger sum takes a day to arrange: that squad comes about 24 hours later (SqCallBtn.txt).
; Calling again while one is on its way or here just says so.
float fNow
float fAt
int iState
int iMin
float fBal
ref rCb

begin Function {}
	let iState := player.AuxVarGetFlt "_wbSqState"
	if iState == 1
		let fNow := GameDaysPassed
		let fAt := player.AuxVarGetFlt "_wbSqAt"
		let iMin := (fAt - fNow) * 1440 + 0.5
		if iMin < 1
			let iMin := 1
		endif
		MessageEx "A courier squad is already on its way: about %.0f minutes." iMin
		return
	elseif iState == 2
		MessageEx "The courier squad is already here."
		return
	endif
	let fBal := player.AuxVarGetFlt "_wbBal"
	let rCb := CompileScript "WastelandBank/SqCallBtn.txt"
	MessageBoxExAlt rCb "^Call a courier squad^An ordinary squad comes in 1 to 2 hours (sooner at higher tiers) with up to 2,500 caps of your balance for withdrawals (5,000 for a Trader, 10,000 for a Mogul). A larger sum takes a day to arrange.%rBalance: %.0f caps.|Call a squad now|Bring 5,000 caps (in a day)|Bring 10,000 caps (in a day)|Bring 25,000 caps (in a day)|Bring my whole balance (in a day)|Cancel" fBal
end
