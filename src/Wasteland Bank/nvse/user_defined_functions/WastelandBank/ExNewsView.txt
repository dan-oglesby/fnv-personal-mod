; Wasteland Exchange: the Mojave Market Wire, newest first (the last 10 headlines).
int iN
int iK
int iSlot
string_var sBody
string_var sLine
int iId
int iDay
ref rHead
ref rCb

begin Function {}
	let iN := player.AuxVarGetFlt "_wxNewsN"
	let sBody := ""
	let iK := 0
	while (iK < 10) && (iK < iN)
		let iSlot := (iN - 1 - iK) % 10
		let iId := player.AuxVarGetFlt "_wxNewsId" iSlot
		let iDay := player.AuxVarGetFlt "_wxNewsDay" iSlot
		let rHead := CompileScript "WastelandBank/ExHeadline.txt"
		let sLine := call rHead iId
		let sBody := sBody + (sv_Construct "Day %.0f: %z%r" iDay sLine)
		let iK += 1
	loop
	if iN == 0
		let sBody := "No market news yet."
	endif
	let rCb := CompileScript "WastelandBank/ExBack.txt"
	MessageBoxExAlt rCb "^Mojave Market Wire^%z|Back" sBody
end
