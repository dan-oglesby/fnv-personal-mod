; Wasteland Exchange: caps per NCR dollar. Vanilla is 0.4 (a $5 note is worth 2 caps); this is
; scaled by the NCR dollar's floating multiplier (ticker 14, NCRD), exactly, not the rounded
; item value.
float fM
float fR
ref rPrice

begin Function {}
	let fR := 0.4
	if eval (player.AuxVarGetFlt "_wxInit") == 1
		let rPrice := CompileScript "WastelandBank/ExPrice.txt"
		let fM := call rPrice 14
		let fR := 0.4 * fM
	endif
	SetFunctionValue fR
end
