; Wasteland Bank: the main menu. Every figure is read live when the menu opens. Cash, goods and
; investments are handled at the bank counters (2026-11-18): any mailbox or Mojave Express dropbox
; (Post.txt). Next to one, the first button opens it; elsewhere it says where to go.
float fBal
float fEarned
float fLoan
int iCarry
ref rCaps
ref rCb
ref rNear
int iNear
string_var sPost

begin Function {}
	let rCaps := BuildRef 0 15	; Caps001
	let iCarry := player.GetItemCount rCaps
	let fBal := player.AuxVarGetFlt "_wbBal"
	let fEarned := player.AuxVarGetFlt "_wbEarned"
	let fLoan := player.AuxVarGetFlt "_wbLoan"
	let rNear := CompileScript "WastelandBank/PostNear.txt"
	let iNear := call rNear
	if iNear == 1
		let sPost := "Bank counter"
	else
		let sPost := "Bank counter (at a mailbox or Mojave Express dropbox)"
	endif
	let rCb := CompileScript "WastelandBank/MainBtn.txt"
	MessageBoxExAlt rCb "^Wasteland Bank^Balance: %.0f caps (interest earned: %.0f)%rLoan owed: %.0f caps%rCarrying: %.0f caps|%z|Wasteland Exchange (stocks)|Settings|Close" fBal fEarned fLoan iCarry sPost
end
