; Wasteland Exchange: account settings (auto-invest mode, reserve, reinvest-all-dividends).
int iMode
float fReserve
string_var sMode
string_var sDrip
string_var sDripBtn
ref rCb

begin Function {}
	let iMode := player.AuxVarGetFlt "_wxAutoMode"
	let sMode := "off"
	if iMode == 1
		let sMode := "index (every listed stock)"
	elseif iMode == 2
		let sMode := "my stocks only"
	endif
	let fReserve := player.AuxVarGetFlt "_wxReserve"
	let sDrip := "off"
	let sDripBtn := "Reinvest all dividends: turn on"
	if eval (player.AuxVarGetFlt "_wxDripAll") == 1
		let sDrip := "on"
		let sDripBtn := "Reinvest all dividends: turn off"
	endif
	let rCb := CompileScript "WastelandBank/ExSettingsBtn.txt"
	MessageBoxExAlt rCb "^Exchange Settings^Auto-invest: %z%rOnce a game day, your bank balance above the reserve is invested in 100-cap steps (up to 10,000 caps a day). Each step buys the stock you hold least of, so it spreads evenly. Index uses every listed stock. My stocks only tops up stocks you already own.%r%rReserve kept in the bank: %.0f caps%r%rReinvest all dividends: %z%rOn: every holding's dividends buy more shares. Off: each stock's own setting applies.|Auto-invest: change mode|Reserve: change amount|%z|Invest now|Back" sMode fReserve sDrip sDripBtn
end
