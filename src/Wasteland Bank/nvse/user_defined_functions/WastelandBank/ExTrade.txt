; Wasteland Exchange: trade through the bank balance. iAct 1 = buy fAmt caps' worth, 2 = sell fAmt
; of your shares (0.5 = half, 1 = all), 3 = short fAmt caps' worth, 4 = cover the whole short.
; Buying and covering pay the ask (price + 1.5%); selling and shorting get the bid (price - 1.5%).
; Shares are fractional. Your own trades don't move the price. Trades go through a courier squad
; (2026-11-16): with none here (SqHere.txt), nothing is traded. Prices can be read any time.
; WBNK (16) has 1,000 shares in all: you can't buy past that, or short it. Buying and selling NCR
; assets (Hub-listed stocks, War Bonds) and the sovereign debt fund moves the currencies a little
; (Flow.txt). Gains and losses on sales count toward this month's profits ("_wbProfM", Tax.txt).
; Shorting needs margin: after the trade, equity (balance + shares - cost to cover shorts) must be
; at least half of all your short positions. Weekly checks cover everything if it drops under 30%.
int iT
int iAct
float fAmt
float fP
float fAsk
float fBid
float fH
float fC
float fBal
float fQ
float fV
float fPart
float fLong
float fShort
float fEq
ref rPrice
ref rTot
ref rCash
ref rHere
int iHere
ref rFlow

begin Function {iT, iAct, fAmt}
	let rHere := CompileScript "WastelandBank/SqHere.txt"
	let iHere := call rHere
	if iHere == 0
		MessageEx "Trades go through the bank's couriers: call a squad from the bank card."
		return
	endif
	if eval (player.AuxVarGetFlt "_wxX" iT) == 1
		MessageEx "That stock is delisted."
		return
	endif
	let rPrice := CompileScript "WastelandBank/ExPrice.txt"
	let rCash := CompileScript "WastelandBank/ExCash.txt"
	let fP := call rPrice iT
	let fAsk := fP * 1.015
	let fBid := fP * 0.985
	let fH := player.AuxVarGetFlt "_wxH" iT
	let fC := player.AuxVarGetFlt "_wxC" iT
	let fBal := player.AuxVarGetFlt "_wbBal"

	if iAct == 1
		if fH < 0
			MessageEx "Cover your short position first."
			return
		endif
		if fAmt > fBal
			let fAmt := fBal
		endif
		if fAmt < 1
			MessageEx "Not enough in the bank. Deposit caps first."
			return
		endif
		let fQ := fAmt / fAsk
		if iT == 16 && fH + fQ > 1000
			let fQ := 1000 - fH
			let fAmt := fQ * fAsk
			if fQ <= 0
				MessageEx "You already own every share of the bank."
				return
			endif
		endif
		let fH += fQ
		let fC += fAmt
		let fV := 0 - fAmt
		call rCash fV
		let rFlow := CompileScript "WastelandBank/Flow.txt"
		call rFlow iT fAmt
		MessageEx "Bought %.2f shares for %.0f caps" fQ fAmt

	elseif iAct == 2
		if fH <= 0
			MessageEx "You don't own any shares of this stock."
			return
		endif
		let fQ := fH * fAmt
		let fV := fQ * fBid
		let fPart := fC * fAmt
		let fH -= fQ
		let fC -= fPart
		if fAmt >= 1
			let fH := 0
			let fC := 0
		endif
		call rCash fV
		let fPart := fV - fPart
		let fLong := player.AuxVarGetFlt "_wbProfM"
		let fLong += fPart
		player.AuxVarSetFlt "_wbProfM" fLong
		let rFlow := CompileScript "WastelandBank/Flow.txt"
		let fLong := 0 - fV
		call rFlow iT fLong
		MessageEx "Sold %.2f shares for %.0f caps (gain %.0f)" fQ fV fPart

	elseif iAct == 3
		if iT == 16
			MessageEx "The bank's shares can't be shorted."
			return
		endif
		if fH > 0
			MessageEx "Sell your shares first, then you can short it."
			return
		endif
		let rTot := CompileScript "WastelandBank/ExTotals.txt"
		let fLong := call rTot 1
		let fShort := call rTot 2
		let fEq := fBal + fLong - fShort
		if fEq < 0.5 * (fShort + fAmt)
			MessageEx "Not enough margin: you need equity of at least half your total short positions."
			return
		endif
		let fQ := fAmt / fBid
		let fH -= fQ
		let fC += fAmt
		call rCash fAmt
		MessageEx "Shorted %.2f shares for %.0f caps" fQ fAmt

	elseif iAct == 4
		if fH >= 0
			MessageEx "You have no short position in this stock."
			return
		endif
		let fQ := 0 - fH
		let fV := fQ * fAsk
		let fPart := fC - fV
		let fH := 0
		let fC := 0
		let fV := 0 - fV
		call rCash fV
		let fLong := player.AuxVarGetFlt "_wbProfM"
		let fLong += fPart
		player.AuxVarSetFlt "_wbProfM" fLong
		MessageEx "Covered %.2f shares (gain %.0f caps)" fQ fPart
	endif

	player.AuxVarSetFlt "_wxH" fH iT
	player.AuxVarSetFlt "_wxC" fC iT
end
