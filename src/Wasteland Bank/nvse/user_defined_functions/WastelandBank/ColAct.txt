; Wasteland Bank: xNVSE OnActivate handler on each debt collector (ColTick.txt), filtered to that
; collector as either argument. They have no dialogue; activating one while they're waiting on you
; shows their demand again (Collect.txt).
ref rFirst
ref rSecond
ref rNext

begin Function {rFirst, rSecond}
	if eval rFirst != player && rSecond != player
		return
	endif
	if eval (player.AuxVarGetFlt "_wbColState") != 2
		return
	endif
	let rNext := CompileScript "WastelandBank/Collect.txt"
	call rNext
end
