; Wasteland Bank: the Legion treasury, through the couriers (2026-11-16). Lend denarii from your
; denarius account (Accounts.txt) for 30 days at 2 percent a week, paid in denarii. (The Legion's
; caravans are backed in person with the Publicanus at Cottonwood Cove, Caravans.txt; they replaced
; the publicani contracts, and contracts already bought run out as before, LegDay.txt.) Only losing the Dam touches the treasury's risk: then 75 percent
; is gone and what's left shrinks to nothing over 30 days; once you collect it, the Legion's
; business with you ends (LegDay.txt, LegionBtn.txt). The treasury deals only with those the Legion
; accepts (LegRep.txt).
; State: loans "_ltA" 0-4 (denarii) ending "_ltEnd" (game day); contracts "_ltC" 0-2 ending
; "_ltCEnd", offered again from "_ltCNext"; "_ltDefault", "_ltOwed", "_ltRec", "_ltClosed".
float fAcct
float fLoans
float fCons
float fRec
int iN
int iK
int iRep
string_var sState
ref rFn
ref rCb
string_var sBtns
int iB

begin Function {}
	let rFn := CompileScript "WastelandBank/LegInit.txt"
	call rFn
	let fAcct := player.AuxVarGetFlt "_wbBalL"
	let fLoans := 0
	let iN := 0
	let iK := 0
	while iK < 5
		let fRec := player.AuxVarGetFlt "_ltA" iK
		if fRec > 0
			let fLoans += fRec
			let iN += 1
		endif
		let iK += 1
	loop
	let fCons := 0
	let iK := 0
	while iK < 3
		let fCons += player.AuxVarGetFlt "_ltC" iK
		let iK += 1
	loop
	let rFn := CompileScript "WastelandBank/LegRep.txt"
	let iRep := call rFn
	let sState := "The treasury deals with you."
	if eval (player.AuxVarGetFlt "_ltClosed") == 1
		let sState := "The Legion's paymaster has fled down the river. Its business here is over."
	elseif eval (player.AuxVarGetFlt "_ltDefault") == 1
		let fRec := player.AuxVarGetFlt "_ltOwed"
		let fRec *= player.AuxVarGetFlt "_ltRec"
		let sState := sv_Construct "The Legion lost the Dam and has defaulted. %.0f denarii can still be recovered, less each day." fRec
	elseif iRep == 0
		let sState := "The treasury won't deal with you: your standing with the Legion is below Accepted."
	elseif iRep == 2
		let sState := "The Legion holds you Vilified: it pays you nothing."
	endif
	; buttons that apply (2026-11-17), noted in "*_ltBtn" for LegionBtn.txt (0 lend 100, 1 lend
	; 1,000, 3 collect after a default, 4 back)
	player.AuxVarErase "*_ltBtn"
	let sBtns := "|Lend 100 denarii for 30 days|Lend 1,000 denarii for 30 days"
	player.AuxVarSetFlt "*_ltBtn" 0 0
	player.AuxVarSetFlt "*_ltBtn" 1 1
	let iB := 2
	if eval (player.AuxVarGetFlt "_ltDefault") == 1 && (player.AuxVarGetFlt "_ltClosed") == 0
		let sBtns := sBtns + "|Collect what can be recovered"
		player.AuxVarSetFlt "*_ltBtn" 3 iB
		let iB += 1
	endif
	let sBtns := sBtns + "|Back"
	player.AuxVarSetFlt "*_ltBtn" 4 iB
	let rCb := CompileScript "WastelandBank/LegionBtn.txt"
	MessageBoxExAlt rCb "^Legion treasury^Your denarius account: %.0f denarii.%rLent: %.0f denarii in %.0f loan(s), 2 percent a week.%rIn publicani contracts: %.0f denarii. The Legion's caravans are backed with the Publicanus at Cottonwood Cove.%r%z%z" fAcct fLoans iN fCons sState sBtns
end
