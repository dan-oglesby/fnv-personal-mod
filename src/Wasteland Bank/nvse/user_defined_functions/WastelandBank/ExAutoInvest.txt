; Wasteland Exchange: auto-invest. Invests the bank balance above the reserve (_wxReserve) in
; 100-cap steps, up to 100 steps (10,000 caps) per run. Run once per game day by ExTick, or by
; "Invest now" in the exchange settings.
; Mode (_wxAutoMode): 1 = index, every listed stock. 2 = my stocks, only stocks already owned.
; Each step buys whichever eligible stock you hold least of by value, so the money spreads
; evenly. Skips delisted stocks, the currencies (not listed) and any stock you are short.
; Buying costs the same as by hand (price + 1.5%).
int iMode
int iT
int iN
int iBest
int iSteps
int iDone
float fBal
float fReserve
float fBest
float fV
float fH
float fP
float fAsk
float fC
float fEx
float fSpent
float fStep
ref rPrice
ref rCash

begin Function {}
	let iMode := player.AuxVarGetFlt "_wxAutoMode"
	if iMode == 0
		return
	endif
	if eval (player.AuxVarGetFlt "_wxInit") == 0
		return
	endif
	if eval (player.AuxVarGetFlt "_wbStage") < 3
		return	; auto-invest opens at the Speculator stage (Stage.txt)
	endif
	let rPrice := CompileScript "WastelandBank/ExPrice.txt"
	let rCash := CompileScript "WastelandBank/ExCash.txt"
	let fReserve := player.AuxVarGetFlt "_wxReserve"
	let iN := player.AuxVarGetFlt "_wxCount"
	let fStep := -100
	let fSpent := 0
	let iSteps := 0
	let iDone := 0
	let fBal := player.AuxVarGetFlt "_wbBal"
	while (iDone == 0) && (iSteps < 100) && (fBal - fReserve >= 100)
		; pick the eligible stock with the smallest holding by value
		let iBest := -1
		let fBest := 0
		let iT := 0
		while iT < iN
			let fEx := player.AuxVarGetFlt "_wxEx" iT
			if fEx <= 2 && iT != 17 && iT != 18
				if eval (player.AuxVarGetFlt "_wxX" iT) == 0
					let fH := player.AuxVarGetFlt "_wxH" iT
					if (fH > 0) || ((iMode == 1) && (fH == 0))
						let fP := call rPrice iT
						let fV := fH * fP
						if (iBest < 0) || (fV < fBest)
							let iBest := iT
							let fBest := fV
						endif
					endif
				endif
			endif
			let iT += 1
		loop
		if iBest < 0
			let iDone := 1
		else
			let fP := call rPrice iBest
			let fAsk := fP * 1.015
			let fH := player.AuxVarGetFlt "_wxH" iBest
			let fH += 100 / fAsk
			player.AuxVarSetFlt "_wxH" fH iBest
			let fC := player.AuxVarGetFlt "_wxC" iBest
			let fC += 100
			player.AuxVarSetFlt "_wxC" fC iBest
			call rCash fStep
			let fBal -= 100
			let fSpent += 100
			let iSteps += 1
		endif
	loop
	if fSpent > 0
		print "Wasteland Exchange: auto-invested " + $fSpent + " caps"
		MessageEx "Wasteland Exchange: auto-invested %.0f caps" fSpent
	endif
	SetFunctionValue fSpent
end
