; Wasteland Exchange: record a news item (by number, see ExHeadline) in the feed (last 10 kept)
; and show it, unless past events are being priced in.
int iId
int iN
int iNext
int iSlot
int iDay
string_var sText
ref rHead

begin Function {iId}
	let iN := player.AuxVarGetFlt "_wxNewsN"
	let iSlot := iN % 10
	let iDay := Floor GameDaysPassed
	player.AuxVarSetFlt "_wxNewsId" iId iSlot
	player.AuxVarSetFlt "_wxNewsDay" iDay iSlot
	let iNext := iN + 1
	player.AuxVarSetFlt "_wxNewsN" iNext
	let rHead := CompileScript "WastelandBank/ExHeadline.txt"
	let sText := call rHead iId
	print "Mojave Market Wire: " + sText
	if eval (player.AuxVarGetFlt "_wxQuiet") == 0
		MessageEx "Mojave Market Wire: %z" sText
	endif
end
