; Wasteland Exchange: add a headline to the news feed (the last 10 are kept) and show it, unless
; past events are being priced in.
string_var sText
string_var sFull
int iN
int iNext
int iSlot
int iDay

begin Function {sText}
	let iN := player.AuxVarGetFlt "_wxNewsN"
	let iSlot := iN % 10
	let iDay := Floor GameDaysPassed
	let sFull := sv_Construct "Day %.0f: %z" iDay sText
	player.AuxVarSetStr "_wxNews" $sFull iSlot
	let iNext := iN + 1
	player.AuxVarSetFlt "_wxNewsN" iNext
	print "Mojave Market Wire: " + sText
	if eval (player.AuxVarGetFlt "_wxQuiet") == 0
		MessageEx "Mojave Market Wire: %z" sText
	endif
end
